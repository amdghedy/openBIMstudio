/**
 * View definitions â€” the single source of truth for every 2D projection.
 *
 * A ViewDef is a *configuration*. It contains no geometry; it describes how
 * to project the element model onto a plane. The generator produces a
 * Scene2D from a ViewDef + the element model. The renderer draws the Scene2D.
 *
 * This separation is what lets plan / section / elevation share the same
 * renderer, annotation engine, and selection system.
 */

export type ViewKind =
  | "plan"
  | "rcp"
  | "section"
  | "elevation"
  | "detail"
  | "drafting"
  | "3d"
  | "schedule"
  | "legend";

export type Discipline = "ARC" | "STR" | "MEP" | "COORD";

export interface PlanConfig {
  levelId: string;
  levelElevation: number;
  cutHeight: number;
  topHeight: number;
  bottomDepth: number;
  viewDepth: number;
}

export interface SectionConfig {
  cutLine: { x1: number; z1: number; x2: number; z2: number };
  lookDir: "left" | "right";
  top: number;
  bottom: number;
  farClip: number;
}

export interface ElevationConfig {
  direction: "north" | "south" | "east" | "west";
  origin: [number, number, number];
  farClip: number;
}

export interface DetailConfig {
  parentViewId: string;
  region: { x: number; y: number; w: number; h: number };
  zoom: number;
}

export interface ViewDef {
  id: string;
  kind: ViewKind;
  name: string;
  scale: number;
  discipline: Discipline;
  templateId?: string;
  crop: { x: number; y: number; w: number; h: number } | null;
  plan?: PlanConfig;
  section?: SectionConfig;
  elevation?: ElevationConfig;
  detail?: DetailConfig;
  createdAt: number;
  updatedAt: number;
}

/**
 * Scene2D â€” renderer-agnostic description of a 2D view.
 *
 * Every view generator produces a Scene2D. Every renderer consumes one.
 * Units are world metres. Stroke widths are paper points â€” converted to
 * world units by paperToWorld() at render time.
 */

export interface StrokeStyle {
  color: string;
  widthPt: number;
  dash: number[] | null;
}

export interface FillStyle {
  color: string;
  opacity: number;
}

export type SceneEntity =
  | { kind: "line"; x1: number; z1: number; x2: number; z2: number; stroke: StrokeStyle }
  | { kind: "polyline"; points: Array<[number, number]>; closed: boolean; stroke: StrokeStyle; fill?: FillStyle }
  | { kind: "arc"; cx: number; cz: number; r: number; a0: number; a1: number; stroke: StrokeStyle }
  | { kind: "circle"; cx: number; cz: number; r: number; stroke: StrokeStyle; fill?: FillStyle }
  | { kind: "text"; x: number; z: number; text: string; anchor: "start" | "middle" | "end"; sizePt: number; rotation: number; color: string }
  | { kind: "symbol"; x: number; z: number; name: string; size: number; rotation: number }
  | { kind: "hatch"; polygon: Array<[number, number]>; pattern: string; color: string; scale: number };

export interface SceneLayer {
  cut: SceneEntity[];
  project: SceneEntity[];
  overhead: SceneEntity[];
  below: SceneEntity[];
  grid: SceneEntity[];
  annotation: SceneEntity[];
  symbols: SceneEntity[];
}

export interface Scene2D {
  bounds: { minX: number; minZ: number; maxX: number; maxZ: number };
  layers: SceneLayer;
  generatedAt: number;
}

export function emptyScene(): Scene2D {
  return {
    bounds: { minX: 0, minZ: 0, maxX: 10, maxZ: 10 },
    layers: { cut: [], project: [], overhead: [], below: [], grid: [], annotation: [], symbols: [] },
    generatedAt: Date.now(),
  };
}

/**
 * Convert paper millimetres to world metres at a given view scale.
 *   paperToWorld(0.18, 100) === 0.018   // 0.18 mm at 1:100 = 18 mm world
 */
export function paperToWorld(paperMm: number, viewScale: number): number {
  return (paperMm / 1000) * viewScale;
}

export function worldToPaper(worldM: number, viewScale: number): number {
  return (worldM / viewScale) * 1000;
}

import type { ViewDef, ViewKind, Discipline } from "./types";

const STORAGE_KEY = "obs.views.v1";

let views: ViewDef[] = [];
const listeners = new Set<() => void>();

function notify(): void {
  for (const cb of listeners) {
    try { cb(); } catch { /* */ }
  }
}

function persist(): void {
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(views)); } catch { /* */ }
}

export function bootViewsStore(): void {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw) {
      const parsed = JSON.parse(raw);
      if (Array.isArray(parsed)) views = parsed;
    }
  } catch { /* */ }
  if (!views.length) seedDefaultViews();
  notify();
}

export function listViews(): ViewDef[] { return views.slice(); }
export function listViewsByKind(kind: ViewKind): ViewDef[] { return views.filter((v) => v.kind === kind); }
export function getView(id: string): ViewDef | null { return views.find((v) => v.id === id) ?? null; }

export function addView(def: ViewDef): ViewDef {
  views.push(def);
  persist();
  notify();
  return def;
}

export function updateView(id: string, patch: Partial<ViewDef>): ViewDef | null {
  const v = views.find((x) => x.id === id);
  if (!v) return null;
  Object.assign(v, patch, { updatedAt: Date.now() });
  persist();
  notify();
  return v;
}

export function deleteView(id: string): boolean {
  const before = views.length;
  views = views.filter((v) => v.id !== id);
  if (views.length === before) return false;
  persist();
  notify();
  return true;
}

export function createView(kind: ViewKind, name: string, discipline: Discipline = "COORD"): ViewDef {
  const now = Date.now();
  const def: ViewDef = {
    id: `view-${now.toString(36)}-${Math.random().toString(36).slice(2, 6)}`,
    kind, name, scale: 100, discipline, crop: null,
    createdAt: now, updatedAt: now,
  };
  if (kind === "plan" || kind === "rcp") {
    def.plan = { levelId: "level-0", levelElevation: 0, cutHeight: 1.2, topHeight: 2.4, bottomDepth: 1.2, viewDepth: 4.0 };
  }
  if (kind === "section") {
    def.section = { cutLine: { x1: -5, z1: 0, x2: 5, z2: 0 }, lookDir: "right", top: 6, bottom: -1, farClip: 30 };
  }
  if (kind === "elevation") {
    def.elevation = { direction: "south", origin: [0, 3, 15], farClip: 60 };
  }
  return addView(def);
}

export function onViewsChange(cb: () => void): () => void {
  listeners.add(cb);
  return () => listeners.delete(cb);
}

/**
 * Plan view generator â€” wraps the existing plan-generator.ts and adapts
 * its PlanView output to the general Scene2D format.
 */

import { generatePlanView } from "../../drawings/plan-generator";
import type { PlanView } from "../../drawings/plan-types";
import type { ViewDef } from "../types";
import type { Scene2D, SceneEntity, StrokeStyle } from "../scene";
import { emptyScene } from "../scene";

function stroke(color: string, widthPt: number, dash: number[] | null = null): StrokeStyle {
  return { color, widthPt, dash };
}

const STYLE = {
  cut:      stroke("#111827", 0.35, null),
  overhead: stroke("#6b7280", 0.18, [1.2, 0.8]),
  below:    stroke("#9ca3af", 0.15, [0.6, 0.8]),
  grid:     stroke("#b91c1c", 0.20, [3, 0.6, 0.6, 0.6]),
};

export async function generatePlanScene(def: ViewDef): Promise<Scene2D> {
  if (!def.plan) return emptyScene();

  const planView: PlanView = await generatePlanView({
    levelName: def.name,
    levelElevation: def.plan.levelElevation,
    cutHeight: def.plan.cutHeight,
    cutDepth: def.plan.bottomDepth,
    viewRangeTop: def.plan.topHeight,
    viewRangeBottom: def.plan.bottomDepth,
    viewDepth: def.plan.viewDepth,
    includeOverhead: true,
    includeBelow: true,
    maxSegments: 250_000,
  });

  const scene = emptyScene();
  scene.bounds = {
    minX: planView.bounds.minX,
    minZ: planView.bounds.minZ,
    maxX: planView.bounds.maxX,
    maxZ: planView.bounds.maxZ,
  };

  for (const seg of planView.segments) {
    const style = (STYLE as any)[seg.kind] ?? STYLE.cut;
    const entity: SceneEntity = {
      kind: "line", x1: seg.x1, z1: seg.z1, x2: seg.x2, z2: seg.z2, stroke: style,
    };
    if (seg.kind === "grid") scene.layers.grid.push(entity);
    else if (seg.kind === "overhead") scene.layers.overhead.push(entity);
    else if (seg.kind === "below") scene.layers.below.push(entity);
    else scene.layers.cut.push(entity);
  }

  for (const arc of planView.arcs) {
    scene.layers.cut.push({
      kind: "arc", cx: arc.cx, cz: arc.cz, r: arc.r, a0: arc.start, a1: arc.end, stroke: STYLE.cut,
    });
  }

  for (const t of planView.texts) {
    scene.layers.symbols.push({
      kind: "text",
      x: t.x, z: t.z,
      text: t.text,
      anchor: t.anchor ?? "middle",
      sizePt: (t.size ?? 0.4) * 10,
      rotation: t.rotation ?? 0,
      color: t.kind === "grid" ? "#a04a4a" : "#1a1a1a",
    });
  }

  scene.generatedAt = Date.now();
  return scene;
}

/**
 * Section view generator â€” analytic vertical cut.
 *
 *   section-2D X = distance along the cut line
 *   section-2D Z = world Y (elevation)
 */

import { getElements } from "../../../features/authoring";
import type { AuthoredElement } from "../../projects/types";
import type { ViewDef } from "../types";
import type { Scene2D, SceneEntity, StrokeStyle } from "../scene";
import { emptyScene } from "../scene";

const CUT_STYLE: StrokeStyle = { color: "#111827", widthPt: 0.5, dash: null };
const PROJECT_STYLE: StrokeStyle = { color: "#6b7280", widthPt: 0.2, dash: null };

interface CutPlane {
  origin: [number, number];
  dir: [number, number];
  normal: [number, number];
}

function makePlane(cut: NonNullable<ViewDef["section"]>): CutPlane {
  const { x1, z1, x2, z2 } = cut.cutLine;
  const dx = x2 - x1;
  const dz = z2 - z1;
  const L = Math.hypot(dx, dz) || 1;
  const dir: [number, number] = [dx / L, dz / L];
  let normal: [number, number] = [-dir[1], dir[0]];
  if (cut.lookDir === "left") normal = [-normal[0], -normal[1]];
  return { origin: [x1, z1], dir, normal };
}

function alongCut(plane: CutPlane, px: number, pz: number): number {
  return (px - plane.origin[0]) * plane.dir[0] + (pz - plane.origin[1]) * plane.dir[1];
}

function signedDist(plane: CutPlane, px: number, pz: number): number {
  return (px - plane.origin[0]) * plane.normal[0] + (pz - plane.origin[1]) * plane.normal[1];
}

export async function generateSectionScene(def: ViewDef): Promise<Scene2D> {
  const scene = emptyScene();
  if (!def.section) return scene;

  const plane = makePlane(def.section);
  const elements = getElements();
  const allX: number[] = [];
  const allZ: number[] = [];

  for (const el of elements) {
    const slice = cutElement(el, plane);
    if (slice) {
      for (const e of slice) { scene.layers.cut.push(e); updateBoundsFromEntity(e, allX, allZ); }
    }
    const projection = projectElement(el, plane, def.section);
    if (projection) {
      for (const e of projection) { scene.layers.project.push(e); updateBoundsFromEntity(e, allX, allZ); }
    }
  }

  if (allX.length) {
    scene.bounds = {
      minX: Math.min(...allX), minZ: Math.min(...allZ),
      maxX: Math.max(...allX), maxZ: Math.max(...allZ),
    };
  }
  scene.generatedAt = Date.now();
  return scene;
}

function cutElement(el: AuthoredElement, plane: CutPlane): SceneEntity[] | null {
  const p = el.params;

  if (p.start && p.end) {
    const [ax, az] = p.start;
    const [bx, bz] = p.end;
    const dA = signedDist(plane, ax, az);
    const dB = signedDist(plane, bx, bz);
    if (Math.sign(dA) !== Math.sign(dB) || Math.abs(dA) < 1e-6 || Math.abs(dB) < 1e-6) {
      const t = dA / (dA - dB);
      const cx = ax + (bx - ax) * t;
      const cz = az + (bz - az) * t;
      const x = alongCut(plane, cx, cz);
      const y0 = p.elevation;
      const y1 = p.elevation + p.height;
      const w = (p.thickness ?? 0.2) / 2;
      return [{
        kind: "polyline", closed: true,
        points: [[x - w, y0], [x + w, y0], [x + w, y1], [x - w, y1]],
        stroke: CUT_STYLE,
        fill: { color: "#cbd5e1", opacity: 0.6 },
      }];
    }
    return null;
  }

  if (p.point) {
    const [px, pz] = p.point;
    const d = signedDist(plane, px, pz);
    const w = (p.thickness ?? 0.3) / 2;
    if (Math.abs(d) > w) return null;
    const x = alongCut(plane, px, pz);
    const y0 = p.elevation;
    const y1 = p.elevation + p.height;
    const halfW = (p.depth ?? p.thickness ?? 0.3) / 2;
    return [{
      kind: "polyline", closed: true,
      points: [[x - halfW, y0], [x + halfW, y0], [x + halfW, y1], [x - halfW, y1]],
      stroke: CUT_STYLE,
      fill: { color: "#cbd5e1", opacity: 0.6 },
    }];
  }

  if (p.polygon && p.polygon.length >= 3) {
    const crossings: number[] = [];
    for (let i = 0; i < p.polygon.length; i++) {
      const a = p.polygon[i];
      const b = p.polygon[(i + 1) % p.polygon.length];
      const dA = signedDist(plane, a[0], a[1]);
      const dB = signedDist(plane, b[0], b[1]);
      if (Math.sign(dA) !== Math.sign(dB)) {
        const t = dA / (dA - dB);
        const cx = a[0] + (b[0] - a[0]) * t;
        const cz = a[1] + (b[1] - a[1]) * t;
        crossings.push(alongCut(plane, cx, cz));
      }
    }
    if (crossings.length >= 2) {
      const x0 = Math.min(...crossings);
      const x1 = Math.max(...crossings);
      const y0 = p.elevation;
      const y1 = p.elevation + p.height;
      return [{
        kind: "polyline", closed: true,
        points: [[x0, y0], [x1, y0], [x1, y1], [x0, y1]],
        stroke: CUT_STYLE,
        fill: { color: "#94a3b8", opacity: 0.6 },
      }];
    }
  }

  return null;
}

function projectElement(
  el: AuthoredElement, plane: CutPlane, cfg: NonNullable<ViewDef["section"]>,
): SceneEntity[] | null {
  const p = el.params;
  const out: SceneEntity[] = [];
  if (p.start && p.end) {
    const [ax, az] = p.start;
    const [bx, bz] = p.end;
    const dA = signedDist(plane, ax, az);
    const dB = signedDist(plane, bx, bz);
    if (dA > 0 && dB > 0 && dA <= cfg.farClip && dB <= cfg.farClip) {
      const x0 = alongCut(plane, ax, az);
      const x1 = alongCut(plane, bx, bz);
      const y0 = p.elevation;
      const y1 = p.elevation + p.height;
      out.push({ kind: "line", x1: x0, z1: y0, x2: x0, z2: y1, stroke: PROJECT_STYLE });
      out.push({ kind: "line", x1: x1, z1: y0, x2: x1, z2: y1, stroke: PROJECT_STYLE });
    }
  }
  return out.length ? out : null;
}

function updateBoundsFromEntity(e: SceneEntity, xs: number[], zs: number[]): void {
  if (e.kind === "line") { xs.push(e.x1, e.x2); zs.push(e.z1, e.z2); }
  else if (e.kind === "polyline") { for (const [x, z] of e.points) { xs.push(x); zs.push(z); } }
  else if (e.kind === "arc") { xs.push(e.cx - e.r, e.cx + e.r); zs.push(e.cz - e.r, e.cz + e.r); }
  else if (e.kind === "circle") { xs.push(e.cx - e.r, e.cx + e.r); zs.push(e.cz - e.r, e.cz + e.r); }
  else if (e.kind === "text" || e.kind === "symbol") { xs.push(e.x); zs.push(e.z); }
}

/**
 * Elevation view generator.
 *
 * Projects authored elements onto a horizontal viewing axis. Produces a
 * silhouette line drawing: each linear element becomes a rectangle outline,
 * each point element becomes a small box.
 */

import { getElements } from "../../../features/authoring";
import type { AuthoredElement } from "../../projects/types";
import type { ViewDef } from "../types";
import type { Scene2D, SceneEntity, StrokeStyle } from "../scene";
import { emptyScene } from "../scene";

const OUTLINE: StrokeStyle = { color: "#111827", widthPt: 0.25, dash: null };

function axisForDir(dir: string): [number, number] {
  switch (dir) {
    case "north": return [1, 0];
    case "south": return [-1, 0];
    case "east":  return [0, 1];
    case "west":  return [0, -1];
    default:      return [1, 0];
  }
}

export async function generateElevationScene(def: ViewDef): Promise<Scene2D> {
  const scene = emptyScene();
  if (!def.elevation) return scene;

  const axis = axisForDir(def.elevation.direction);
  const elements = getElements();
  const xs: number[] = [];
  const zs: number[] = [];

  for (const el of elements) {
    const shapes = projectElement(el, axis);
    if (!shapes) continue;
    for (const s of shapes) {
      scene.layers.project.push(s);
      if (s.kind === "line") { xs.push(s.x1, s.x2); zs.push(s.z1, s.z2); }
      else if (s.kind === "polyline") { for (const [x, z] of s.points) { xs.push(x); zs.push(z); } }
    }
  }

  if (xs.length) {
    scene.bounds = {
      minX: Math.min(...xs), minZ: Math.min(...zs),
      maxX: Math.max(...xs), maxZ: Math.max(...zs),
    };
  }
  scene.generatedAt = Date.now();
  return scene;
}

function projectElement(el: AuthoredElement, axis: [number, number]): SceneEntity[] | null {
  const p = el.params;
  const out: SceneEntity[] = [];

  if (p.start && p.end) {
    const uA = p.start[0] * axis[0] + p.start[1] * axis[1];
    const uB = p.end[0] * axis[0] + p.end[1] * axis[1];
    const y0 = p.elevation;
    const y1 = p.elevation + p.height;
    out.push({
      kind: "polyline", closed: true,
      points: [[uA, y0], [uB, y0], [uB, y1], [uA, y1]],
      stroke: OUTLINE,
    });
  } else if (p.point) {
    const uC = p.point[0] * axis[0] + p.point[1] * axis[1];
    const w = (p.thickness ?? 0.3) / 2;
    const y0 = p.elevation;
    const y1 = p.elevation + p.height;
    out.push({
      kind: "polyline", closed: true,
      points: [[uC - w, y0], [uC + w, y0], [uC + w, y1], [uC - w, y1]],
      stroke: OUTLINE,
    });
  } else if (p.polygon && p.polygon.length >= 3) {
    const us = p.polygon.map(([x, z]) => x * axis[0] + z * axis[1]);
    const u0 = Math.min(...us);
    const u1 = Math.max(...us);
    const y0 = p.elevation;
    const y1 = p.elevation + p.height;
    out.push({
      kind: "polyline", closed: true,
      points: [[u0, y0], [u1, y0], [u1, y1], [u0, y1]],
      stroke: OUTLINE,
    });
  }

  return out.length ? out : null;
}

/**
 * SVG renderer for Scene2D.
 */

import type { Scene2D, SceneEntity, StrokeStyle } from "../scene";
import { paperToWorld } from "../scene";

const SVG_NS = "http://www.w3.org/2000/svg";

export interface SvgRenderOptions {
  scale: number;
  padding: number;
  widthPx: number;
  heightPx: number;
}

export function renderSceneToSvg(scene: Scene2D, opts: SvgRenderOptions): SVGSVGElement {
  const svg = document.createElementNS(SVG_NS, "svg");
  svg.setAttribute("xmlns", SVG_NS);
  svg.setAttribute("width", String(opts.widthPx));
  svg.setAttribute("height", String(opts.heightPx));

  const pad = opts.padding;
  const vbMinX = scene.bounds.minX - pad;
  const vbMinY = -(scene.bounds.maxZ + pad);
  const vbW = (scene.bounds.maxX - scene.bounds.minX) + 2 * pad;
  const vbH = (scene.bounds.maxZ - scene.bounds.minZ) + 2 * pad;
  svg.setAttribute("viewBox", `${vbMinX} ${vbMinY} ${vbW} ${vbH}`);

  const order: Array<[string, SceneEntity[]]> = [
    ["below", scene.layers.below],
    ["grid", scene.layers.grid],
    ["project", scene.layers.project],
    ["overhead", scene.layers.overhead],
    ["cut", scene.layers.cut],
    ["symbols", scene.layers.symbols],
    ["annotation", scene.layers.annotation],
  ];

  for (const [name, entities] of order) {
    const g = document.createElementNS(SVG_NS, "g");
    g.setAttribute("data-layer", name);
    for (const e of entities) drawEntity(g, e, opts.scale);
    svg.appendChild(g);
  }

  return svg;
}

function drawEntity(parent: SVGGElement, e: SceneEntity, viewScale: number): void {
  switch (e.kind) {
    case "line": {
      const el = document.createElementNS(SVG_NS, "line");
      el.setAttribute("x1", String(e.x1));
      el.setAttribute("y1", String(-e.z1));
      el.setAttribute("x2", String(e.x2));
      el.setAttribute("y2", String(-e.z2));
      applyStroke(el, e.stroke, viewScale);
      parent.appendChild(el);
      break;
    }
    case "polyline": {
      const d = e.points.map((p, i) => `${i === 0 ? "M" : "L"}${p[0]} ${-p[1]}`).join(" ") + (e.closed ? " Z" : "");
      const el = document.createElementNS(SVG_NS, "path");
      el.setAttribute("d", d);
      if (e.fill) {
        el.setAttribute("fill", e.fill.color);
        el.setAttribute("fill-opacity", String(e.fill.opacity));
      } else el.setAttribute("fill", "none");
      applyStroke(el, e.stroke, viewScale);
      parent.appendChild(el);
      break;
    }
    case "arc": {
      const cx = e.cx;
      const cy = -e.cz;
      const r = e.r;
      const x0 = cx + Math.cos(e.a0) * r;
      const y0 = cy - Math.sin(e.a0) * r;
      const x1 = cx + Math.cos(e.a1) * r;
      const y1 = cy - Math.sin(e.a1) * r;
      const large = Math.abs(e.a1 - e.a0) > Math.PI ? 1 : 0;
      const sweep = e.a1 > e.a0 ? 0 : 1;
      const el = document.createElementNS(SVG_NS, "path");
      el.setAttribute("d", `M ${x0} ${y0} A ${r} ${r} 0 ${large} ${sweep} ${x1} ${y1}`);
      el.setAttribute("fill", "none");
      applyStroke(el, e.stroke, viewScale);
      parent.appendChild(el);
      break;
    }
    case "circle": {
      const el = document.createElementNS(SVG_NS, "circle");
      el.setAttribute("cx", String(e.cx));
      el.setAttribute("cy", String(-e.cz));
      el.setAttribute("r", String(e.r));
      el.setAttribute("fill", e.fill ? e.fill.color : "none");
      if (e.fill) el.setAttribute("fill-opacity", String(e.fill.opacity));
      applyStroke(el, e.stroke, viewScale);
      parent.appendChild(el);
      break;
    }
    case "text": {
      const el = document.createElementNS(SVG_NS, "text");
      el.setAttribute("x", String(e.x));
      el.setAttribute("y", String(-e.z));
      el.setAttribute("text-anchor", e.anchor);
      el.setAttribute("dominant-baseline", "middle");
      el.setAttribute("fill", e.color);
      el.setAttribute("font-family", "Arial, system-ui, sans-serif");
      el.setAttribute("font-size", String(paperToWorld(e.sizePt, viewScale)));
      if (e.rotation) el.setAttribute("transform", `rotate(${e.rotation} ${e.x} ${-e.z})`);
      el.textContent = e.text;
      parent.appendChild(el);
      break;
    }
    default:
      break;
  }
}

function applyStroke(el: SVGElement, stroke: StrokeStyle, viewScale: number): void {
  el.setAttribute("stroke", stroke.color);
  el.setAttribute("stroke-width", String(paperToWorld(stroke.widthPt, viewScale)));
  el.setAttribute("stroke-linecap", "round");
  el.setAttribute("stroke-linejoin", "round");
  if (stroke.dash) {
    el.setAttribute("stroke-dasharray", stroke.dash.map((d) => paperToWorld(d, viewScale)).join(" "));
  }
}

/**
 * Dimension engine.
 *
 * A dimension in the model is a config object. This module turns it into a
 * concrete Scene2D: witness lines, dimension line, tick marks, and text.
 *
 * Every dimension references elements through the Reference system, so when
 * a wall moves the dimension re-resolves automatically. See references.ts.
 */

import type { AuthoredElement } from "../projects/types";
import { resolveReference } from "./references";
import type { Dimension, DimensionStyle } from "./annotations";
import { DEFAULT_DIM_STYLES } from "./annotations";
import type { SceneEntity } from "./scene";
import { paperToWorld } from "./scene";

export interface ResolvedDimension {
  entities: SceneEntity[];
  text: string;
  value: number;
  valid: boolean;
}

export function getDimensionStyle(dim: Dimension): DimensionStyle {
  return DEFAULT_DIM_STYLES.find((s) => s.id === dim.styleId) ?? DEFAULT_DIM_STYLES[0];
}

export function formatDimensionValue(meters: number, style: DimensionStyle): string {
  const factor = style.unit === "m" ? 1 : style.unit === "cm" ? 100 : 1000;
  const value = meters * factor;
  return value.toFixed(style.precision);
}

export function resolveDimension(
  dim: Dimension,
  elements: readonly AuthoredElement[],
  viewScale: number,
): ResolvedDimension {
  const style = getDimensionStyle(dim);
  const p1 = resolveReference(dim.ref1, elements);

  if (!p1) {
    return { entities: [], text: "?", value: 0, valid: false };
  }

  // Single-point tools
  if (dim.kind === "spot-elevation" || dim.kind === "spot-coordinate") {
    return resolveSpot(dim, p1, style, viewScale);
  }

  const p2 = dim.ref2 ? resolveReference(dim.ref2, elements) : null;
  if (!p2) {
    return { entities: [], text: "?", value: 0, valid: false };
  }

  if (dim.kind === "angular") {
    return resolveAngular(dim, p1, p2, style, viewScale);
  }
  if (dim.kind === "radial" || dim.kind === "diameter") {
    return resolveRadial(dim, p1, p2, style, viewScale);
  }
  // aligned | linear | arc share the straight-line renderer
  return resolveLinear(dim, p1, p2, style, viewScale);
}

function resolveSpot(
  dim: Dimension,
  p: [number, number, number],
  style: DimensionStyle,
  viewScale: number,
): ResolvedDimension {
  const size = paperToWorld(style.textSizePt, viewScale);
  const leaderLen = paperToWorld(12, viewScale);
  const tickR = paperToWorld(1.2, viewScale);
  const entities: SceneEntity[] = [];

  entities.push({
    kind: "circle", cx: p[0], cz: p[2], r: tickR,
    stroke: { color: style.color, widthPt: 0.18, dash: null },
  });
  entities.push({
    kind: "line",
    x1: p[0], z1: p[2],
    x2: p[0], z2: p[2] + leaderLen,
    stroke: { color: style.color, widthPt: 0.18, dash: null },
  });

  const text = dim.override
    ?? (dim.kind === "spot-elevation"
      ? `EL ${formatDimensionValue(p[1], style)}`
      : `X ${formatDimensionValue(p[0], style)}  Z ${formatDimensionValue(p[2], style)}`);

  entities.push({
    kind: "text",
    x: p[0],
    z: p[2] + leaderLen + size,
    text,
    anchor: "middle",
    sizePt: style.textSizePt,
    rotation: 0,
    color: style.color,
  });

  return { entities, text, value: dim.kind === "spot-elevation" ? p[1] : 0, valid: true };
}

function resolveAngular(
  dim: Dimension,
  p1: [number, number, number],
  p2: [number, number, number],
  style: DimensionStyle,
  viewScale: number,
): ResolvedDimension {
  const vertex = dim.vertexRef
    ? resolveReference(dim.vertexRef, [])
    : null;
  const vx = vertex ? vertex[0] : (p1[0] + p2[0]) / 2;
  const vz = vertex ? vertex[2] : (p1[2] + p2[2]) / 2;

  const a1 = Math.atan2(p1[2] - vz, p1[0] - vx);
  const a2 = Math.atan2(p2[2] - vz, p2[0] - vx);
  let deg = Math.abs((a2 - a1) * 180 / Math.PI);
  if (deg > 180) deg = 360 - deg;

  const r = paperToWorld(14, viewScale);
  const arcPath: SceneEntity = {
    kind: "arc",
    cx: vx, cz: vz, r,
    a0: a1, a1: a2,
    stroke: { color: style.color, widthPt: 0.2, dash: null },
  };
  const text = dim.override ?? `${deg.toFixed(1)}\u00B0`;
  const mid = (a1 + a2) / 2;
  const tx = vx + Math.cos(mid) * (r + paperToWorld(3, viewScale));
  const tz = vz + Math.sin(mid) * (r + paperToWorld(3, viewScale));
  const textEntity: SceneEntity = {
    kind: "text",
    x: tx, z: tz,
    text,
    anchor: "middle",
    sizePt: style.textSizePt,
    rotation: 0,
    color: style.color,
  };

  return { entities: [arcPath, textEntity], text, value: deg, valid: true };
}

function resolveRadial(
  dim: Dimension,
  p1: [number, number, number],
  p2: [number, number, number],
  style: DimensionStyle,
  viewScale: number,
): ResolvedDimension {
  const dx = p2[0] - p1[0];
  const dz = p2[2] - p1[2];
  const r = Math.hypot(dx, dz);

  const entities: SceneEntity[] = [{
    kind: "line",
    x1: p1[0], z1: p1[2],
    x2: p2[0], z2: p2[2],
    stroke: { color: style.color, widthPt: 0.2, dash: null },
  }];

  const text = dim.override ?? (dim.kind === "radial"
    ? `R ${formatDimensionValue(r, style)}`
    : `\u00D8 ${formatDimensionValue(r * 2, style)}`);

  const mx = (p1[0] + p2[0]) / 2;
  const mz = (p1[2] + p2[2]) / 2;
  entities.push({
    kind: "text",
    x: mx, z: mz + paperToWorld(3, viewScale),
    text,
    anchor: "middle",
    sizePt: style.textSizePt,
    rotation: 0,
    color: style.color,
  });

  return { entities, text, value: r, valid: true };
}

function resolveLinear(
  dim: Dimension,
  p1: [number, number, number],
  p2: [number, number, number],
  style: DimensionStyle,
  viewScale: number,
): ResolvedDimension {
  // In 2D views the cut plane is Xâ€“Z. For plan, use X and Z directly. For
  // section, the generators map (along, Y) â†’ (x, z). The dimension engine is
  // view-agnostic â€” it works on the view's 2D coordinates, so the caller is
  // expected to have already swapped the reference to view coordinates when
  // it created the dimension.

  // Force horizontal or vertical for "linear"
  let x2 = p2[0];
  let z2 = p2[2];
  if (dim.kind === "linear") {
    if (Math.abs(x2 - p1[0]) >= Math.abs(z2 - p1[2])) z2 = p1[2];
    else x2 = p1[0];
  }

  const dx = x2 - p1[0];
  const dz = z2 - p1[2];
  const L = Math.hypot(dx, dz);
  if (L < 1e-9) {
    return { entities: [], text: "?", value: 0, valid: false };
  }

  const ux = dx / L;
  const uz = dz / L;
  const nx = -uz;
  const nz = ux;
  const off = paperToWorld(dim.offsetMm, viewScale);

  const a0x = p1[0] + nx * off;
  const a0z = p1[2] + nz * off;
  const a1x = x2 + nx * off;
  const a1z = z2 + nz * off;

  const stroke = { color: style.color, widthPt: 0.2, dash: null };
  const entities: SceneEntity[] = [];

  // Witness lines from each referenced point to the dim line
  entities.push({
    kind: "line",
    x1: p1[0], z1: p1[2],
    x2: a0x, z2: a0z,
    stroke,
  });
  entities.push({
    kind: "line",
    x1: x2, z1: z2,
    x2: a1x, z2: a1z,
    stroke,
  });

  // Dimension line
  entities.push({
    kind: "line",
    x1: a0x, z1: a0z,
    x2: a1x, z2: a1z,
    stroke,
  });

  // Tick marks at each end
  const tickLen = paperToWorld(2.5, viewScale);
  for (const [px, pz] of [[a0x, a0z], [a1x, a1z]] as Array<[number, number]>) {
    const t1x = px - ux * tickLen * 0.3 - nx * tickLen;
    const t1z = pz - uz * tickLen * 0.3 - nz * tickLen;
    const t2x = px + ux * tickLen * 0.3 + nx * tickLen;
    const t2z = pz + uz * tickLen * 0.3 + nz * tickLen;
    entities.push({
      kind: "line",
      x1: t1x, z1: t1z,
      x2: t2x, z2: t2z,
      stroke: { color: style.color, widthPt: 0.25, dash: null },
    });
  }

  const text = dim.override ?? formatDimensionValue(L, style);
  const mx = (a0x + a1x) / 2 + nx * paperToWorld(2, viewScale);
  const mz = (a0z + a1z) / 2 + nz * paperToWorld(2, viewScale);
  entities.push({
    kind: "text",
    x: mx, z: mz,
    text,
    anchor: "middle",
    sizePt: style.textSizePt,
    rotation: 0,
    color: style.color,
  });

  return { entities, text, value: L, valid: true };
}

/**
 * Tag engine.
 *
 * A tag is a reference + a field selector. This module reads the field from
 * the referenced element and produces a Scene2D bubble + optional leader.
 */

import type { AuthoredElement } from "../projects/types";
import { resolveReference } from "./references";
import type { Tag } from "./annotations";
import type { SceneEntity } from "./scene";
import { paperToWorld } from "./scene";

export interface ResolvedTag {
  entities: SceneEntity[];
  text: string;
  valid: boolean;
}

const TAG_COLOR = "#0f766e";

export function tagText(
  tag: Tag,
  el: AuthoredElement | null,
): string {
  if (tag.override) return tag.override;
  if (!el) return "\u2014";

  switch (tag.kind) {
    case "category": return el.name;
    case "material": return String(el.properties?.material ?? "\u2014");
    case "room": {
      const n = el.properties?.roomNumber;
      const nm = el.properties?.roomName;
      if (n && nm) return `${n} \u00B7 ${nm}`;
      if (n) return String(n);
      if (nm) return String(nm);
      return "\u2014";
    }
    case "custom": {
      const k = tag.field;
      if (!k) return "\u2014";
      return String(el.properties?.[k] ?? el.params[k as keyof typeof el.params] ?? "\u2014");
    }
  }
}

export function resolveTag(
  tag: Tag,
  elements: readonly AuthoredElement[],
  viewScale: number,
): ResolvedTag {
  const el = elements.find((e) => e.id === tag.ref.elementId) ?? null;
  const text = tagText(tag, el);

  const anchor = resolveReference(tag.ref, elements);
  if (!anchor) {
    return { entities: [], text, valid: false };
  }

  const [ax, , az] = anchor;
  const offX = paperToWorld(tag.offsetMm[0], viewScale);
  const offZ = paperToWorld(tag.offsetMm[1], viewScale);
  const tx = ax + offX;
  const tz = az + offZ;

  const entities: SceneEntity[] = [];

  if (tag.leader && tag.leaderShape !== "none") {
    entities.push({
      kind: "line",
      x1: ax, z1: az,
      x2: tx, z2: tz,
      stroke: { color: TAG_COLOR, widthPt: 0.18, dash: null },
    });
  }

  // Bubble rectangle sized by text length
  const size = paperToWorld(3.0, viewScale);
  const charW = size * 0.55;
  const w = Math.max(size, text.length * charW) + paperToWorld(2, viewScale);
  const h = size * 1.4;

  entities.push({
    kind: "polyline",
    closed: true,
    points: [
      [tx - w / 2, tz - h / 2],
      [tx + w / 2, tz - h / 2],
      [tx + w / 2, tz + h / 2],
      [tx - w / 2, tz + h / 2],
    ],
    stroke: { color: TAG_COLOR, widthPt: 0.15, dash: null },
    fill: { color: "#ecfdf5", opacity: 0.85 },
  });

  entities.push({
    kind: "text",
    x: tx, z: tz,
    text,
    anchor: "middle",
    sizePt: 3.0,
    rotation: tag.rotation,
    color: TAG_COLOR,
  });

  return { entities, text, valid: true };
}

/**
 * Sheet layout â€” transforms between scene (world metre) coordinates and
 * sheet millimeter coordinates.
 *
 * A viewport on a sheet has an x/y/width/height in mm. The view inside it
 * is rendered at the viewport's paper scale. The world viewBox width is:
 *
 *   world_width_m = paper_width_mm * scaleDenom / 1000
 */

import type { SheetDef, SheetViewport } from "../projects/types";
import type { ViewDef } from "./types";
import type { Scene2D } from "./scene";

export interface ViewportTransform {
  x: number;
  y: number;
  widthMm: number;
  heightMm: number;
  scaleDenom: number;
  viewBox: { x: number; y: number; w: number; h: number };
}

export function parseScale(s: string | number | undefined, fallback = 100): number {
  if (typeof s === "number" && Number.isFinite(s) && s > 0) return s;
  if (typeof s === "string") {
    const m = s.match(/1\s*:\s*(\d+(?:\.\d+)?)/);
    if (m) return parseFloat(m[1]);
    const n = parseFloat(s);
    if (Number.isFinite(n) && n > 0) return n;
  }
  return fallback;
}

export function computeViewportTransform(
  viewport: SheetViewport,
  sheet: SheetDef,
  view: ViewDef,
  scene: Scene2D,
): ViewportTransform {
  const scaleDenom = parseScale(viewport.scale, view.scale);
  const widthMm = viewport.width;
  const heightMm = viewport.height;
  const worldW = (widthMm * scaleDenom) / 1000;
  const worldH = (heightMm * scaleDenom) / 1000;

  const cx = (scene.bounds.minX + scene.bounds.maxX) / 2;
  const cz = (scene.bounds.minZ + scene.bounds.maxZ) / 2;
  const vbX = cx - worldW / 2;
  const vbY = -cz - worldH / 2;

  void sheet;
  return {
    x: viewport.x,
    y: viewport.y,
    widthMm,
    heightMm,
    scaleDenom,
    viewBox: { x: vbX, y: vbY, w: worldW, h: worldH },
  };
}

export function formatScale(denom: number): string {
  return `1:${Math.round(denom)}`;
}

/**
 * Plan view generator.
 *
 * Cut plane at `levelElevation + cutHeight`. Looks down (bird's eye).
 * Fragment geometry: per-element triangle edges, filtered to the cut band,
 * projected to XZ, deduplicated at 1 mm resolution.
 *
 * Authored elements (walls, doors, windows, columns, beams, slabs) generate
 * parametric 2D symbols â€” real Revit-style plan representations, not
 * bounding boxes.
 *
 * Screen convention inside the plan: `screenX = worldX`, `screenY = -worldZ`.
 * The panel handles the flip when rendering SVG.
 */
import * as THREE from "three";
import { ctx } from "../../core/context";
import { getElementTriangles } from "../clash/geometry";
import {
  getElements,
  getGrids,
  getLevels,
} from "../../features/authoring";
import type { AuthoredElement } from "../projects/types";
import type {
  PlanArc, PlanGenerationOptions, PlanSegment, PlanText, PlanView,
} from "./plan-types";

const EDGE_ROUND = 1000; // 1 mm dedup
const MIN_EDGE_LEN = 0.005; // 5 mm â€” kill degenerate projections

function key2(x: number, z: number): string {
  return `${Math.round(x * EDGE_ROUND)},${Math.round(z * EDGE_ROUND)}`;
}

function edgeKey(x1: number, z1: number, x2: number, z2: number): string {
  const a = key2(x1, z1);
  const b = key2(x2, z2);
  return a < b ? `${a}|${b}` : `${b}|${a}`;
}

export async function generatePlanView(
  opts: PlanGenerationOptions,
): Promise<PlanView> {
  // Revit-like view range: Top is above the cut plane; Bottom and View Depth
  // control how far below the associated level the plan can see. Legacy cutDepth
  // remains the fallback for callers that only provide the original API.
  const lowDepth = Math.max(
    Math.abs(opts.viewRangeBottom ?? 0),
    Math.abs(opts.viewDepth ?? opts.cutDepth),
  );
  const highDepth = Math.max(opts.cutHeight, Math.abs(opts.viewRangeTop ?? opts.cutHeight));
  const bandLow = opts.levelElevation - lowDepth;
  const bandHigh = opts.levelElevation + highDepth;
  const cutElevation = opts.levelElevation + opts.cutHeight;

  const segments: PlanSegment[] = [];
  const arcs: PlanArc[] = [];
  const texts: PlanText[] = [];
  let truncated = false;
  const edgeSeen = new Set<string>();

  const pushSeg = (s: PlanSegment) => {
    if (segments.length >= opts.maxSegments) {
      truncated = true;
      return;
    }
    const len = Math.hypot(s.x2 - s.x1, s.z2 - s.z1);
    if (len < MIN_EDGE_LEN) return;
    segments.push(s);
  };

  // â”€â”€ 1. Authored elements (all modeling categories) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  const authored = getElements();
  // Include every authored element; symbol layer chooses cut / overhead / below
  const authoredInBand = authored.filter((el) => {
    const elev = el.params.elevation ?? 0;
    const h = el.params.height ?? 0.1;
    // Always include linear MEP / structure even if slightly outside band
    if (["duct", "pipe", "cable-tray", "conduit", "beam", "brace", "truss", "mline", "strip"].includes(el.type)) {
      return elev <= bandHigh + 3 && elev + h >= bandLow - 1;
    }
    return elev <= bandHigh && elev + h >= bandLow;
  });

  // Build door/window map by host wall
  const hostedByWall = new Map<string, AuthoredElement[]>();
  for (const el of authored) {
    if ((el.type === "door" || el.type === "window") && el.hostId) {
      if (!hostedByWall.has(el.hostId)) hostedByWall.set(el.hostId, []);
      hostedByWall.get(el.hostId)!.push(el);
    }
  }

  for (const el of authoredInBand) {
    if (opts.signal?.cancelled) break;
    emitAuthoredSymbol(el, segments, arcs, texts, pushSeg, hostedByWall, cutElevation, bandLow, bandHigh);
  }

  // â”€â”€ 2. Fragments (triangle edge extraction) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  opts.onProgress?.({ phase: "Reading models", current: 0, total: ctx.fragments.list.size });

  let mi = 0;
  let fragmentElements = 0;
  let fragmentEdgeCount = 0;
  for (const [modelId, model] of ctx.fragments.list) {
    mi++;
    opts.onProgress?.({ phase: `Model: ${modelId}`, current: mi, total: ctx.fragments.list.size });
    if (opts.signal?.cancelled) break;

    let ids: number[] = [];
    let boxes: THREE.Box3[] = [];
    try {
      ids = (await model.getItemsIdsWithGeometry()) ?? [];
      boxes = await model.getBoxes(ids);
    } catch {
      continue;
    }

    try {
      model.object.updateMatrixWorld(true);
    } catch { /* */ }
    const mw: THREE.Matrix4 = model.object.matrixWorld?.clone?.() ?? new THREE.Matrix4();

    for (let i = 0; i < ids.length; i++) {
      if (opts.signal?.cancelled) break;
      if (truncated && segments.length >= opts.maxSegments) break;

      const b = boxes[i];
      if (!b || b.isEmpty()) continue;
      const world = b.clone().applyMatrix4(mw);
      // Quick reject: box outside the cut band
      if (world.max.y < bandLow || world.min.y > bandHigh) continue;

      fragmentElements++;

      let tris: any[] | null = null;
      try {
        tris = getElementTriangles(model.object, ids[i]);
      } catch {
        tris = null;
      }

      if (!tris || !tris.length) {
        // Fallback â€” footprint of the bounding box
        emitBoxFootprint(world, pushSeg, "cut");
        continue;
      }

      for (const t of tris) {
        const y0 = t[1], y1 = t[4], y2 = t[7];
        const minY = Math.min(y0, y1, y2);
        const maxY = Math.max(y0, y1, y2);
        if (maxY < bandLow || minY > bandHigh) continue;

        const verts = [
          [t[0], t[1], t[2]],
          [t[3], t[4], t[5]],
          [t[6], t[7], t[8]],
        ];
        for (let e = 0; e < 3; e++) {
          const a = verts[e];
          const b2 = verts[(e + 1) % 3];
          const k = edgeKey(a[0], a[2], b2[0], b2[2]);
          if (edgeSeen.has(k)) continue;
          edgeSeen.add(k);
          fragmentEdgeCount++;
          pushSeg({ x1: a[0], z1: a[2], x2: b2[0], z2: b2[2], kind: "cut" });
        }
      }
    }
    await new Promise((r) => setTimeout(r, 0));
  }

  // â”€â”€ 3. Grid lines â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  for (const gr of getGrids()) {
    // Extend grid to just past the plan bounds (will be clamped later)
    const dx = gr.end[0] - gr.start[0];
    const dz = gr.end[1] - gr.start[1];
    const len = Math.hypot(dx, dz);
    if (len < 0.01) continue;
    const ux = dx / len, uz = dz / len;
    const pad = 3.0;
    const x1 = gr.start[0] - ux * pad;
    const z1 = gr.start[1] - uz * pad;
    const x2 = gr.end[0] + ux * pad;
    const z2 = gr.end[1] + uz * pad;
    pushSeg({ x1, z1, x2, z2, kind: "grid" });
    // Bubble at end
    texts.push({
      x: x2 + ux * 0.5,
      z: z2 + uz * 0.5,
      text: gr.label,
      kind: "grid",
      anchor: "middle",
      size: 0.6,
    });
  }

  // â”€â”€ 4. Level text annotation â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  texts.push({
    x: 0, z: 0,
    text: `${opts.levelName}  (+${opts.levelElevation.toFixed(2)} m)`,
    kind: "level",
    anchor: "start",
    size: 0.5,
  });

  // â”€â”€ 5. Compute bounds â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
  let minX = Infinity, minZ = Infinity, maxX = -Infinity, maxZ = -Infinity;
  const consider = (x: number, z: number) => {
    if (x < minX) minX = x;
    if (x > maxX) maxX = x;
    if (z < minZ) minZ = z;
    if (z > maxZ) maxZ = z;
  };
  for (const s of segments) { consider(s.x1, s.z1); consider(s.x2, s.z2); }
  for (const a of arcs) { consider(a.cx - a.r, a.cz - a.r); consider(a.cx + a.r, a.cz + a.r); }
  if (!Number.isFinite(minX)) { minX = 0; minZ = 0; maxX = 10; maxZ = 10; }

  return {
    level: { name: opts.levelName, elevation: opts.levelElevation },
    cutElevation,
    segments,
    arcs,
    texts,
    bounds: { minX, minZ, maxX, maxZ },
    generatedAt: Date.now(),
    truncated,
    stats: {
      authoredElements: authoredInBand.length,
      fragmentElements,
      segments: segments.length,
      fragmentEdgeCount,
    },
  };
}

// â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•
// Authored element plan symbols
// â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•â•

function emitLinearThick(
  push: (s: PlanSegment) => void,
  el: AuthoredElement,
  kind: "cut" | "overhead" | "below",
  thickness: number,
): void {
  const p = el.params;
  if (!p.start || !p.end) return;
  const [ax, az] = p.start;
  const [bx, bz] = p.end;
  const dx = bx - ax, dz = bz - az;
  const L = Math.hypot(dx, dz);
  if (L < 0.02) return;
  const ux = dx / L, uz = dz / L;
  const nx = -uz, nz = ux;
  const t2 = Math.max(0.02, thickness / 2);
  push({ x1: ax + nx * t2, z1: az + nz * t2, x2: bx + nx * t2, z2: bz + nz * t2, kind, source: el.id });
  push({ x1: ax - nx * t2, z1: az - nz * t2, x2: bx - nx * t2, z2: bz - nz * t2, kind, source: el.id });
  push({ x1: ax + nx * t2, z1: az + nz * t2, x2: ax - nx * t2, z2: az - nz * t2, kind, source: el.id });
  push({ x1: bx + nx * t2, z1: bz + nz * t2, x2: bx - nx * t2, z2: bz - nz * t2, kind, source: el.id });
}

function emitAuthoredSymbol(
  el: AuthoredElement,
  segments: PlanSegment[],
  arcs: PlanArc[],
  texts: PlanText[],
  push: (s: PlanSegment) => void,
  hostedByWall: Map<string, AuthoredElement[]>,
  cutElevation: number,
  bandLow = 0,
  bandHigh = 3,
): void {
  const p = el.params;
  const elev = p.elevation ?? 0;
  const elevKind: "cut" | "overhead" | "below" =
    elev > cutElevation + 0.05 ? "overhead" : elev + (p.height ?? 0) < bandLow ? "below" : "cut";

  switch (el.type) {
    // miter-linear-v4
    // Only wall / beam / brace / strip share the mitered plan footprint.
    // MEP types use flat thickness so the corner is filled by the elbow
    // fitting â€” matching Revit.
    case "wall":
    case "beam":
    case "brace":
    case "strip": {
      // Revit-style linear join v2 - robust thickness, wider tolerance,
      // perpendicularity preference, wider miter cap. Applies to walls
      // and beams; both miter identically at L-corners.
      if (!p.start || !p.end) return;
      const [ax, az] = p.start;
      const [bx, bz] = p.end;
      const dx = bx - ax, dz = bz - az;
      const L = Math.hypot(dx, dz);
      if (L < 0.05) return;
      const ux = dx / L, uz = dz / L;
      const nx = -uz, nz = ux;
      const rawThick = Number(p.thickness);
      const wallThick = Number.isFinite(rawThick) && rawThick > 0.001 ? rawThick : 0.2;
      const t2 = wallThick / 2;
      const joinStat = String((el as any).properties?.joinStatus ?? "allow");

      const endTol = Math.max(0.20, t2 * 4);

      const findCornerPartner = (px: number, pz: number): AuthoredElement | null => {
        if (joinStat === "disallow" || joinStat === "butt") return null;
        let best: AuthoredElement | null = null;
        let bestScore = -Infinity;
        for (const o of getElements()) {
          if (o.id === el.id || o.type !== el.type || !o.params.start || !o.params.end) continue;
          const otherJoin = String((o as any).properties?.joinStatus ?? "allow");
          if (otherJoin === "disallow" || otherJoin === "butt") continue;
          const dS = Math.hypot(o.params.start[0] - px, o.params.start[1] - pz);
          const dE = Math.hypot(o.params.end[0] - px, o.params.end[1] - pz);
          const dMin = Math.min(dS, dE);
          if (dMin > endTol) continue;
          const odx = o.params.end[0] - o.params.start[0];
          const odz = o.params.end[1] - o.params.start[1];
          const oL = Math.hypot(odx, odz) || 1;
          const oux = odx / oL, ouz = odz / oL;
          const cross = Math.abs(ux * ouz - uz * oux);
          const score = cross * 100 - dMin * 5;
          if (score > bestScore) { bestScore = score; best = o; }
        }
        return best;
      };

      const lineHit = (
        px: number, pz: number, dx_: number, dz_: number,
        qx: number, qz: number, ex_: number, ez_: number,
      ): { x: number; z: number } | null => {
        const det = dx_ * ez_ - dz_ * ex_;
        if (Math.abs(det) < 1e-9) return null;
        const t = ((qx - px) * ez_ - (qz - pz) * ex_) / det;
        return { x: px + t * dx_, z: pz + t * dz_ };
      };

      const faceEndsAt = (
        endX: number, endZ: number,
      ): { left: { x: number; z: number }; right: { x: number; z: number }; joined: boolean } => {
        const plainL = { x: endX + nx * t2, z: endZ + nz * t2 };
        const plainR = { x: endX - nx * t2, z: endZ - nz * t2 };
        const nb = findCornerPartner(endX, endZ);
        if (!nb || !nb.params.start || !nb.params.end) {
          return { left: plainL, right: plainR, joined: false };
        }
        const [sx, sz] = nb.params.start;
        const [ex, ez] = nb.params.end;
        const ndx = ex - sx, ndz = ez - sz;
        const nL = Math.hypot(ndx, ndz) || 1;
        const nux = ndx / nL, nuz = ndz / nL;
        const nnx = -nuz, nnz = nux;
        const nbThick = Number(nb.params.thickness);
        const nt2 = Math.max(0.01, (Number.isFinite(nbThick) && nbThick > 0.001 ? nbThick : 0.2) / 2);

        const nLeftX = sx + nnx * nt2, nLeftZ = sz + nnz * nt2;
        const nRightX = sx - nnx * nt2, nRightZ = sz - nnz * nt2;

        const LL = lineHit(plainL.x, plainL.z, ux, uz, nLeftX, nLeftZ, nux, nuz);
        const LR = lineHit(plainL.x, plainL.z, ux, uz, nRightX, nRightZ, nux, nuz);
        const RL = lineHit(plainR.x, plainR.z, ux, uz, nLeftX, nLeftZ, nux, nuz);
        const RR = lineHit(plainR.x, plainR.z, ux, uz, nRightX, nRightZ, nux, nuz);

        // miter-pair-LL-RR-v4
        // Revit-style miter: for a perpendicular L-corner, the outer
        // corner is always on the same "side" for both walls (both left
        // faces, or both right faces). Therefore the correct pairing is
        // ALWAYS left â†’ LL, right â†’ RR. The previous turn-sign logic
        // mis-paired right-hand turns (e.g. wall going east meeting a
        // wall going south).
        const maxMiter = Math.max(t2, nt2) * 8;
        const leftCandidate  = LL;
        const rightCandidate = RR;

        const leftOK  = !!leftCandidate  && Math.hypot(leftCandidate.x  - endX, leftCandidate.z  - endZ) <= maxMiter;
        const rightOK = !!rightCandidate && Math.hypot(rightCandidate.x - endX, rightCandidate.z - endZ) <= maxMiter;

        return {
          left:  leftOK  ? leftCandidate!  : plainL,
          right: rightOK ? rightCandidate! : plainR,
          joined: true,
        };
      };

      const startFaces = faceEndsAt(ax, az);
      const endFaces = faceEndsAt(bx, bz);

      const hosts = (hostedByWall.get(el.id) ?? []).filter(
        (h) => h.type === "door" || h.type === "window",
      );
      const openings: { start: number; end: number; el: AuthoredElement }[] = [];
      for (const h of hosts) {
        const w = h.params.width ?? 0.9;
        const off = h.params.offsetAlongHost ?? L / 2;
        openings.push({ start: Math.max(0, off - w / 2), end: Math.min(L, off + w / 2), el: h });
      }
      openings.sort((a, b) => a.start - b.start);

      const merged: { start: number; end: number; el?: AuthoredElement }[] = [];
      for (const o of openings) {
        if (merged.length && o.start <= merged[merged.length - 1].end + 0.01) {
          merged[merged.length - 1].end = Math.max(merged[merged.length - 1].end, o.end);
        } else {
          merged.push({ ...o });
        }
      }

      const emitSides = (from: number, to: number) => {
        if (to - from < 0.02) return;
        const t0 = from / L, t1 = to / L;
        const l1x = startFaces.left.x + (endFaces.left.x - startFaces.left.x) * t0;
        const l1z = startFaces.left.z + (endFaces.left.z - startFaces.left.z) * t0;
        const l2x = startFaces.left.x + (endFaces.left.x - startFaces.left.x) * t1;
        const l2z = startFaces.left.z + (endFaces.left.z - startFaces.left.z) * t1;
        const r1x = startFaces.right.x + (endFaces.right.x - startFaces.right.x) * t0;
        const r1z = startFaces.right.z + (endFaces.right.z - startFaces.right.z) * t0;
        const r2x = startFaces.right.x + (endFaces.right.x - startFaces.right.x) * t1;
        const r2z = startFaces.right.z + (endFaces.right.z - startFaces.right.z) * t1;
        push({ x1: l1x, z1: l1z, x2: l2x, z2: l2z, kind: elevKind, source: el.id });
        push({ x1: r1x, z1: r1z, x2: r2x, z2: r2z, kind: elevKind, source: el.id });
      };

      let cursor = 0;
      for (const o of merged) {
        emitSides(cursor, o.start);
        cursor = o.end;
      }
      emitSides(cursor, L);

      if (!startFaces.joined) {
        push({
          x1: startFaces.left.x, z1: startFaces.left.z,
          x2: startFaces.right.x, z2: startFaces.right.z,
          kind: elevKind, source: el.id,
        });
      }
      if (!endFaces.joined) {
        push({
          x1: endFaces.left.x, z1: endFaces.left.z,
          x2: endFaces.right.x, z2: endFaces.right.z,
          kind: elevKind, source: el.id,
        });
      }

      for (const o of merged) {
        const openingEl = openings.find((x) => x.start === o.start)?.el;
        if (openingEl?.type === "door") {
          const hingeX = ax + ux * o.start;
          const hingeZ = az + uz * o.start;
          const w = o.end - o.start;
          arcs.push({
            cx: hingeX, cz: hingeZ, r: w,
            start: Math.atan2(nz, nx),
            end: Math.atan2(uz, ux),
            kind: "door-swing",
          });
          push({
            x1: hingeX, z1: hingeZ,
            x2: hingeX + nx * w, z2: hingeZ + nz * w,
            kind: "cut", source: openingEl.id,
          });
        } else if (openingEl?.type === "window") {
          const l1x = ax + ux * o.start, l1z = az + uz * o.start;
          const l2x = ax + ux * o.end, l2z = az + uz * o.end;
          push({ x1: l1x + nx * t2, z1: l1z + nz * t2, x2: l1x - nx * t2, z2: l1z - nz * t2, kind: "cut" });
          push({ x1: l2x + nx * t2, z1: l2z + nz * t2, x2: l2x - nx * t2, z2: l2z - nz * t2, kind: "cut" });
        }
      }
      break;
    }
    case "column":
    case "equipment":
    case "footing": {
      if (!p.point) return;
      const w = p.thickness;
      const d = p.depth ?? p.thickness;
      const x = p.point[0], z = p.point[1];
      const hw = w / 2, hd = d / 2;
      push({ x1: x - hw, z1: z - hd, x2: x + hw, z2: z - hd, kind: "cut", source: el.id });
      push({ x1: x + hw, z1: z - hd, x2: x + hw, z2: z + hd, kind: "cut", source: el.id });
      push({ x1: x + hw, z1: z + hd, x2: x - hw, z2: z + hd, kind: "cut", source: el.id });
      push({ x1: x - hw, z1: z + hd, x2: x - hw, z2: z - hd, kind: "cut", source: el.id });
      break;
    }

    case "slab": {
      if (!p.polygon || p.polygon.length < 3) return;
      const poly = p.polygon;
      for (let i = 0; i < poly.length; i++) {
        const a = poly[i];
        const b = poly[(i + 1) % poly.length];
        push({ x1: a[0], z1: a[1], x2: b[0], z2: b[1], kind: elevKind, source: el.id });
      }
      break;
    }

    case "truss":
    case "mline":
      emitLinearThick(push, el, elevKind, p.thickness ?? 0.2);
      break;

    // MEP categories â€” flat thickness, fitting covers the corner
    case "duct":
      emitLinearThick(push, el, elevKind, p.thickness ?? 0.6);
      break;
    case "pipe":
      emitLinearThick(push, el, elevKind, Math.max(0.08, p.thickness ?? 0.1));
      break;
    case "cable-tray":
      emitLinearThick(push, el, elevKind, p.thickness ?? 0.3);
      break;
    case "conduit":
      emitLinearThick(push, el, elevKind, Math.max(0.05, p.thickness ?? 0.05));
      break;
    case "fitting": {
      // MEP elbow v1: render the fitting as two concentric arcs matching
      // the 3D buildFitting geometry. Revit draws duct/pipe elbows as
      // clean quarter (or half) circles in plan â€” not the tiny box the
      // previous placeholder produced.
      const cxF = p.point?.[0];
      const czF = p.point?.[1];
      if (cxF == null || czF == null) {
        // Legacy line-based fitting fallback: small box.
        if (p.start && p.end) {
          const x = (p.start[0] + p.end[0]) / 2;
          const z = (p.start[1] + p.end[1]) / 2;
          const s = (p.thickness ?? 0.3) / 2;
          push({ x1: x - s, z1: z - s, x2: x + s, z2: z - s, kind: elevKind, source: el.id });
          push({ x1: x + s, z1: z - s, x2: x + s, z2: z + s, kind: elevKind, source: el.id });
          push({ x1: x + s, z1: z + s, x2: x - s, z2: z + s, kind: elevKind, source: el.id });
          push({ x1: x - s, z1: z + s, x2: x - s, z2: z - s, kind: elevKind, source: el.id });
        }
        break;
      }

      const parentType = String((p as any).parentType || "duct");
      const kind = String((p as any).fittingKind || "elbow");
      const aIn = p.angleIn ?? 0;
      const aOut = p.angleOut ?? Math.PI / 2;
      const t = p.thickness ?? 0.3;
      const t2 = Math.max(0.02, t / 2);
      const isRound = parentType === "pipe" || parentType === "conduit";
      // Enforce Revit-style minimum: centerline bend radius >= duct width.
      const ductWidth = Math.max(t, 0.05);
      const baseR = Number(p.bendRadius);
      const r = Math.max(ductWidth, Number.isFinite(baseR) && baseR > 0
        ? baseR
        : ductWidth);

      if (kind === "elbow") {
        // MEP elbow v2 â€” correct circular-arc construction.
        //
        // Geometry, in order:
        //   â€¢ Corner C is where the two centerlines meet.
        //   â€¢ Tangent points T1, T2 sit r units back along each duct:
        //         T1 = C âˆ’ dir(aIn) * r
        //         T2 = C + dir(aOut) * r
        //   â€¢ Bend center O is r units from T1 perpendicular to aIn,
        //     on the side that faces the outgoing duct. Because it is
        //     also r units from T2 perpendicular to aOut, the two
        //     perpendiculars intersect exactly at O.
        //   â€¢ The outer boundary arc is at radius r + t/2 from O,
        //     the inner boundary at r âˆ’ t/2.
        const inX  = cxF - Math.cos(aIn)  * r;
        const inZ  = czF - Math.sin(aIn)  * r;
        const outX = cxF + Math.cos(aOut) * r;
        const outZ = czF + Math.sin(aOut) * r;

        // Perpendicular unit vectors to aIn:
        const perpA = { x: Math.sin(aIn), z: -Math.cos(aIn) };
        const perpB = { x: -Math.sin(aIn), z: Math.cos(aIn) };
        // Pick the perpendicular that points from T1 toward T2.
        const toOut = { x: outX - inX, z: outZ - inZ };
        const useA = (toOut.x * perpA.x + toOut.z * perpA.z) > 0;
        const perp = useA ? perpA : perpB;

        const CbX = inX + perp.x * r;
        const CbZ = inZ + perp.z * r;

        // Angles on the bend circle. Normalise sweep to (âˆ’Ï€, Ï€].
        const inletAngle = Math.atan2(inZ - CbZ, inX - CbX);
        let sweep = aOut - aIn;
        while (sweep > Math.PI)  sweep -= 2 * Math.PI;
        while (sweep < -Math.PI) sweep += 2 * Math.PI;
        const arcEnd = inletAngle + sweep;

        const rOuter = r + t2;
        const rInner = Math.max(0.02, r - t2);

        arcs.push({
          cx: CbX, cz: CbZ, r: rOuter,
          start: inletAngle, end: arcEnd,
          kind: "fitting",
        });
        arcs.push({
          cx: CbX, cz: CbZ, r: rInner,
          start: inletAngle, end: arcEnd,
          kind: "fitting",
        });
        break;
      }

      // Tee / cross / cap / transition: small box fallback
      const s = t / 2;
      push({ x1: cxF - s, z1: czF - s, x2: cxF + s, z2: czF - s, kind: elevKind, source: el.id });
      push({ x1: cxF + s, z1: czF - s, x2: cxF + s, z2: czF + s, kind: elevKind, source: el.id });
      push({ x1: cxF + s, z1: czF + s, x2: cxF - s, z2: czF + s, kind: elevKind, source: el.id });
      push({ x1: cxF - s, z1: czF + s, x2: cxF - s, z2: czF - s, kind: elevKind, source: el.id });
      break;
    }

    case "door":
    case "window":
      // Handled by host wall projection
      break;

    default: {
      // Any other category: linear if start/end, point if point, else skip
      if (p.start && p.end) {
        emitLinearThick(push, el, elevKind, p.thickness ?? 0.2);
      } else if (p.point) {
        const w = p.thickness ?? 0.3;
        const d = p.depth ?? w;
        const x = p.point[0], z = p.point[1];
        const hw = w / 2, hd = d / 2;
        push({ x1: x - hw, z1: z - hd, x2: x + hw, z2: z - hd, kind: elevKind, source: el.id });
        push({ x1: x + hw, z1: z - hd, x2: x + hw, z2: z + hd, kind: elevKind, source: el.id });
        push({ x1: x + hw, z1: z + hd, x2: x - hw, z2: z + hd, kind: elevKind, source: el.id });
        push({ x1: x - hw, z1: z + hd, x2: x - hw, z2: z - hd, kind: elevKind, source: el.id });
      } else if (p.polygon && p.polygon.length >= 2) {
        for (let i = 0; i < p.polygon.length; i++) {
          const a = p.polygon[i];
          const b = p.polygon[(i + 1) % p.polygon.length];
          push({ x1: a[0], z1: a[1], x2: b[0], z2: b[1], kind: elevKind, source: el.id });
        }
      }
      break;
    }
  }
}

/**
 * DXF ASCII writer (AutoCAD R2010 / AC1024 compatible subset).
 * Pure TypeScript - no native deps.
 */

import type { DxfDocument, DxfEntity, DxfLayer, DxfPoint } from "./dxf-types";

function pair(code: number, value: string | number): string {
  return `${code}\n${value}\n`;
}

function writePoint(codes: [number, number, number], p: DxfPoint): string {
  return (
    pair(codes[0], p.x) +
    pair(codes[1], p.y) +
    pair(codes[2], p.z ?? 0)
  );
}

function layerTable(layers: DxfLayer[]): string {
  let s = "";
  s += pair(0, "TABLE");
  s += pair(2, "LAYER");
  s += pair(5, "2");
  s += pair(100, "AcDbSymbolTable");
  s += pair(70, layers.length);
  for (const layer of layers) {
    s += pair(0, "LAYER");
    s += pair(5, Math.floor(Math.random() * 0xfffff).toString(16));
    s += pair(100, "AcDbSymbolTableRecord");
    s += pair(100, "AcDbLayerTableRecord");
    s += pair(2, layer.name);
    s += pair(70, layer.frozen ? 1 : 0);
    s += pair(62, layer.color || 7);
    s += pair(6, layer.lineType || "Continuous");
  }
  s += pair(0, "ENDTAB");
  return s;
}

function writeEntity(e: DxfEntity): string {
  let s = "";
  const layer = e.layer || "0";
  const color = e.color ?? 256; // ByLayer

  if (e.type === "LINE") {
    s += pair(0, "LINE");
    s += pair(8, layer);
    s += pair(62, color);
    s += writePoint([10, 20, 30], e.start);
    s += writePoint([11, 21, 31], e.end);
  } else if (e.type === "CIRCLE") {
    s += pair(0, "CIRCLE");
    s += pair(8, layer);
    s += pair(62, color);
    s += writePoint([10, 20, 30], e.center);
    s += pair(40, e.radius);
  } else if (e.type === "ARC") {
    s += pair(0, "ARC");
    s += pair(8, layer);
    s += pair(62, color);
    s += writePoint([10, 20, 30], e.center);
    s += pair(40, e.radius);
    s += pair(50, e.startAngle);
    s += pair(51, e.endAngle);
  } else if (e.type === "LWPOLYLINE") {
    s += pair(0, "LWPOLYLINE");
    s += pair(8, layer);
    s += pair(62, color);
    s += pair(90, e.points.length);
    s += pair(70, e.closed ? 1 : 0);
    for (const p of e.points) {
      s += pair(10, p.x);
      s += pair(20, p.y);
    }
  } else if (e.type === "TEXT") {
    s += pair(0, "TEXT");
    s += pair(8, layer);
    s += pair(62, color);
    s += writePoint([10, 20, 30], e.position);
    s += pair(40, e.height);
    s += pair(1, e.value);
    if (e.rotation != null) s += pair(50, e.rotation);
  } else if (e.type === "POINT") {
    s += pair(0, "POINT");
    s += pair(8, layer);
    s += pair(62, color);
    s += writePoint([10, 20, 30], e.position);
  }
  return s;
}

function computeExtents(entities: DxfEntity[]): { min: DxfPoint; max: DxfPoint } {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  const add = (p: DxfPoint) => {
    minX = Math.min(minX, p.x);
    minY = Math.min(minY, p.y);
    maxX = Math.max(maxX, p.x);
    maxY = Math.max(maxY, p.y);
  };
  for (const e of entities) {
    if (e.type === "LINE") {
      add(e.start);
      add(e.end);
    } else if (e.type === "CIRCLE" || e.type === "ARC") {
      add({ x: e.center.x - e.radius, y: e.center.y - e.radius });
      add({ x: e.center.x + e.radius, y: e.center.y + e.radius });
    } else if (e.type === "LWPOLYLINE") {
      e.points.forEach(add);
    } else if (e.type === "TEXT" || e.type === "POINT") {
      add(e.position);
    }
  }
  if (!Number.isFinite(minX)) {
    return { min: { x: 0, y: 0 }, max: { x: 100, y: 100 } };
  }
  return { min: { x: minX, y: minY }, max: { x: maxX, y: maxY } };
}

/** Serialize a document to DXF R2010 ASCII. */
export function writeDxf(doc: DxfDocument): string {
  const layers =
    doc.layers.length > 0
      ? doc.layers
      : [{ name: "0", color: 7 }];
  const extents = doc.extents ?? computeExtents(doc.entities);

  let s = "";
  // HEADER
  s += pair(0, "SECTION");
  s += pair(2, "HEADER");
  s += pair(9, "$ACADVER");
  s += pair(1, "AC1024");
  s += pair(9, "$INSUNITS");
  const unitsMap: Record<string, number> = {
    unitless: 0,
    in: 1,
    ft: 2,
    mm: 4,
    m: 6,
  };
  s += pair(70, unitsMap[doc.units ?? "mm"] ?? 4);
  s += pair(9, "$EXTMIN");
  s += writePoint([10, 20, 30], extents.min);
  s += pair(9, "$EXTMAX");
  s += writePoint([10, 20, 30], extents.max);
  s += pair(0, "ENDSEC");

  // TABLES
  s += pair(0, "SECTION");
  s += pair(2, "TABLES");
  // LTYPE
  s += pair(0, "TABLE");
  s += pair(2, "LTYPE");
  s += pair(5, "1");
  s += pair(100, "AcDbSymbolTable");
  s += pair(70, 1);
  s += pair(0, "LTYPE");
  s += pair(5, "10");
  s += pair(100, "AcDbSymbolTableRecord");
  s += pair(100, "AcDbLinetypeTableRecord");
  s += pair(2, "Continuous");
  s += pair(70, 0);
  s += pair(3, "Solid line");
  s += pair(72, 65);
  s += pair(73, 0);
  s += pair(40, 0);
  s += pair(0, "ENDTAB");
  s += layerTable(layers);
  s += pair(0, "ENDSEC");

  // ENTITIES
  s += pair(0, "SECTION");
  s += pair(2, "ENTITIES");
  for (const e of doc.entities) {
    s += writeEntity(e);
  }
  s += pair(0, "ENDSEC");

  s += pair(0, "EOF");
  return s;
}

/** Download a DXF string as a file in the browser. */
export function downloadDxf(doc: DxfDocument, filename = "drawing.dxf"): void {
  const text = writeDxf(doc);
  const blob = new Blob([text], { type: "application/dxf" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename.endsWith(".dxf") ? filename : `${filename}.dxf`;
  a.click();
  URL.revokeObjectURL(url);
}

/** ACI index - hex (subset). */
export function aciToHex(aci: number): string {
  const table: Record<number, string> = {
    1: "#ff0000",
    2: "#ffff00",
    3: "#00ff00",
    4: "#00ffff",
    5: "#0000ff",
    6: "#ff00ff",
    7: "#ffffff",
    8: "#808080",
    9: "#c0c0c0",
  };
  if (table[aci]) return table[aci];
  // AutoCAD ACI approximate
  const idx = ((aci % 255) + 255) % 255;
  const h = (idx * 37) % 360;
  return `hsl(${h} 70% 55%)`;
}

export function cadToDxfDocument(): DxfDocument {
  const layers = cadEngine.listLayers().map((l, i) => ({
    name: l.name,
    color: 7 - (i % 6),
  }));
  if (!layers.length) layers.push({ name: "0", color: 7 });

  const entities: DxfEntity[] = [];
  for (const e of cadEngine.getAllEntities()) {
    const layer = cadEngine.listLayers().find((l) => l.id === e.layer)?.name ?? "0";
    if (e.kind === "line" && e.b) {
      entities.push({
        type: "LINE",
        layer,
        start: { x: e.a.x, y: -e.a.y },
        end: { x: e.b.x, y: -e.b.y },
      });
    } else if (e.kind === "rect" && e.b) {
      const x0 = e.a.x, y0 = -e.a.y, x1 = e.b.x, y1 = -e.b.y;
      entities.push({
        type: "LWPOLYLINE",
        layer,
        closed: true,
        points: [
          { x: x0, y: y0 },
          { x: x1, y: y0 },
          { x: x1, y: y1 },
          { x: x0, y: y1 },
        ],
      });
    } else if (e.kind === "circle" && e.r != null) {
      entities.push({
        type: "CIRCLE",
        layer,
        center: { x: e.a.x, y: -e.a.y },
        radius: e.r,
      });
    } else if (e.kind === "arc" && e.r != null && e.startAngle != null && e.endAngle != null) {
      // SVG/screen Y is flipped; angles need care - export degrees
      entities.push({
        type: "ARC",
        layer,
        center: { x: e.a.x, y: -e.a.y },
        radius: e.r,
        startAngle: (-e.endAngle * 180) / Math.PI,
        endAngle: (-e.startAngle * 180) / Math.PI,
      });
    } else if (e.kind === "polyline" && e.points) {
      entities.push({
        type: "LWPOLYLINE",
        layer,
        points: e.points.map((p) => ({ x: p.x, y: -p.y })),
        closed: false,
      });
    } else if (e.kind === "text") {
      entities.push({
        type: "TEXT",
        layer,
        position: { x: e.a.x, y: -e.a.y },
        height: 14,
        value: e.text ?? "",
      });
    }
  }

  return {
    layers,
    entities,
    units: "mm",
    sourceFormat: "cad-engine",
  };
}

export function loadDxfIntoCad(
  doc: DxfDocument,
  opts: { clear?: boolean; scale?: number; offset?: CadPt } = {},
): number {
  const scale = opts.scale ?? 1;
  const ox = opts.offset?.x ?? 40;
  const oy = opts.offset?.y ?? 40;
  if (opts.clear !== false) cadEngine.clear();

  // Ensure layers
  for (const l of doc.layers) {
    const existing = cadEngine.listLayers().find((x) => x.name === l.name);
    if (!existing) {
      cadEngine.addLayer(l.name, aciToHex(l.color));
    }
  }

  const mapLayer = (name: string): string => {
    const found = cadEngine.listLayers().find((l) => l.name === name);
    if (found) {
      cadEngine.setActiveLayer(found.id);
      return found.id;
    }
    return cadEngine.getActiveLayer().id;
  };

  const toScreen = (p: DxfPoint): CadPt => ({
    x: p.x * scale + ox,
    y: -p.y * scale + oy, // DXF Y-up - screen Y-down
  });

  let n = 0;
  for (const e of doc.entities) {
    mapLayer(e.layer);
    if (e.type === "LINE") {
      cadEngine.addLine(toScreen(e.start), toScreen(e.end));
      n++;
    } else if (e.type === "CIRCLE") {
      cadEngine.addCircle(toScreen(e.center), e.radius * scale);
      n++;
    } else if (e.type === "ARC") {
      const c = toScreen(e.center);
      const a0 = (-e.startAngle * Math.PI) / 180;
      const a1 = (-e.endAngle * Math.PI) / 180;
      cadEngine.addArc(c, e.radius * scale, a0, a1);
      n++;
    } else if (e.type === "LWPOLYLINE") {
      const pts = e.points.map(toScreen);
      if (e.closed && pts.length) pts.push({ ...pts[0] });
      cadEngine.addPolyline(pts);
      n++;
    } else if (e.type === "TEXT") {
      cadEngine.addText(toScreen(e.position), e.value);
      n++;
    }
  }
  return n;
}

/* -- THREE preview group --------------------------------------------------- */

function ensureThreeGroup(): THREE.Group | null {
  const scene = (ctx as Any).world?.scene?.three as THREE.Scene | undefined;
  if (!scene) return null;
  if (!threeGroup) {
    threeGroup = new THREE.Group();
    threeGroup.name = "obs-drawings-dxf";
    scene.add(threeGroup);
  }
  return threeGroup;
}

export function loadDxfIntoScene(doc: DxfDocument, opts: { scale?: number } = {}): number {
  const group = ensureThreeGroup();
  if (!group) return 0;
  while (group.children.length) {
    const ch = group.children[0];
    group.remove(ch);
    (ch as Any).geometry?.dispose?.();
  }

  const scale = opts.scale ?? 0.001; // mm - m default
  const mat = new THREE.LineBasicMaterial({ color: 0x38bdf8 });
  let n = 0;

  const addLine = (a: DxfPoint, b: DxfPoint, color?: number) => {
    const geo = new THREE.BufferGeometry().setFromPoints([
      new THREE.Vector3(a.x * scale, (a.z ?? 0) * scale, -a.y * scale),
      new THREE.Vector3(b.x * scale, (b.z ?? 0) * scale, -b.y * scale),
    ]);
    const m = color != null ? new THREE.LineBasicMaterial({ color: new THREE.Color(aciToHex(color)) }) : mat;
    group!.add(new THREE.Line(geo, m));
    n++;
  };

  for (const e of doc.entities) {
    if (e.type === "LINE") addLine(e.start, e.end, e.color);
    else if (e.type === "CIRCLE") {
      const curve = new THREE.EllipseCurve(
        0, 0, e.radius, e.radius, 0, Math.PI * 2, false, 0,
      );
      const pts = curve.getPoints(48).map(
        (p) =>
          new THREE.Vector3(
            (e.center.x + p.x) * scale,
            (e.center.z ?? 0) * scale,
            -(e.center.y + p.y) * scale,
          ),
      );
      const geo = new THREE.BufferGeometry().setFromPoints(pts);
      group.add(new THREE.LineLoop(geo, mat));
      n++;
    } else if (e.type === "LWPOLYLINE") {
      for (let i = 0; i < e.points.length - 1; i++) {
        addLine(e.points[i], e.points[i + 1], e.color);
      }
      if (e.closed && e.points.length > 1) {
        addLine(e.points[e.points.length - 1], e.points[0], e.color);
      }
    } else if (e.type === "ARC") {
      const a0 = (e.startAngle * Math.PI) / 180;
      const a1 = (e.endAngle * Math.PI) / 180;
      const curve = new THREE.EllipseCurve(
        0, 0, e.radius, e.radius, a0, a1, false, 0,
      );
      const pts = curve.getPoints(32).map(
        (p) =>
          new THREE.Vector3(
            (e.center.x + p.x) * scale,
            (e.center.z ?? 0) * scale,
            -(e.center.y + p.y) * scale,
          ),
      );
      const geo = new THREE.BufferGeometry().setFromPoints(pts);
      group.add(new THREE.Line(geo, mat));
      n++;
    }
  }
  return n;
}

}

    for (const e of ents) {
      const type = String(e.type ?? e.entityType ?? "").toUpperCase();
      const layer = e.layer ?? e.layerName ?? "0";
      if (type.includes("LINE") && e.start && e.end) {
        entities.push({
          type: "LINE",
          layer,
          start: { x: e.start.x, y: e.start.y, z: e.start.z },
          end: { x: e.end.x, y: e.end.y, z: e.end.z },
        });
      } else if (type.includes("CIRCLE") && e.center && e.radius != null) {
        entities.push({
          type: "CIRCLE",
          layer,
          center: { x: e.center.x, y: e.center.y },
          radius: e.radius,
        });
      }
    }

    const doc: DxfDocument = {
      layers,
      entities,
      sourceFormat: "dwg",
    };
    const id = uid();
    const drawing: DrawingDocument = {
      id,
      name: file.name,
      format: "dwg",
      doc,
      loadedAt: Date.now(),
    };
    drawings.set(id, drawing);
    activeId = id;
    loadDxfIntoCad(doc);
    loadDxfIntoScene(doc);
    toast("success", `DWG imported: ${entities.length} entities`);
    return drawing;
  } catch (err) {
    console.warn("[dwg-engine] DWG import unavailable:", err);
    return null;
  }
}

export async function importDrawingFile(file: File): Promise<DrawingDocument | null> {
  const name = file.name.toLowerCase();
  if (name.endsWith(".dxf")) {
    return importDxfFile(file, { intoCad: true, intoScene: true });
  }
  if (name.endsWith(".dwg")) {
    const result = await tryImportDwg(file);
    if (result) return result;
    toast(
      "warn",
      "DWG needs @mlightcad/libredwg-web - export DXF from CAD, or: npm i @mlightcad/libredwg-web",
    );
    return null;
  }
  toast("warn", "Use .dxf or .dwg files");
  return null;
}

import type { SheetDef, SheetViewport, TitleblockDef } from "../projects/types";

const uid = () =>
  (crypto as any).randomUUID?.() ?? `sh-${Date.now()}-${Math.random().toString(16).slice(2)}`;

const SHEET_SIZES_MM: Record<SheetDef["size"], { w: number; h: number }> = {
  A0: { w: 1189, h: 841 },
  A1: { w: 841, h: 594 },
  A2: { w: 594, h: 420 },
  A3: { w: 420, h: 297 },
  A4: { w: 297, h: 210 },
  "ANSI-D": { w: 864, h: 559 },
  "ANSI-E": { w: 1118, h: 864 },
  "Arch-D": { w: 914, h: 610 },
  "Arch-E": { w: 1219, h: 914 },
};

let sheets: SheetDef[] = [];
let titleblocks: TitleblockDef[] = [];
const listeners = new Set<() => void>();

function notify() {
  for (const fn of listeners) fn();
}

export function onSheetsChange(fn: () => void) {
  listeners.add(fn);
  return () => listeners.delete(fn);
}

export function listSheets(): SheetDef[] {
  return sheets.slice().sort((a, b) => a.number.localeCompare(b.number, undefined, { numeric: true }));
}

export function getSheet(id: string): SheetDef | undefined {
  return sheets.find((s) => s.id === id);
}

export function createSheet(opts?: Partial<Pick<SheetDef, "number" | "name" | "size" | "orientation">>): SheetDef {
  const n = sheets.length + 1;
  const sheet: SheetDef = {
    id: uid(),
    number: opts?.number ?? `A-${String(n).padStart(3, "0")}`,
    name: opts?.name ?? `Sheet ${n}`,
    size: opts?.size ?? "A1",
    orientation: opts?.orientation ?? "landscape",
    viewports: [],
    annotations: [],
    createdAt: Date.now(),
    updatedAt: Date.now(),
  };
  // Attach default titleblock if available
  const tb = titleblocks[0];
  if (tb) sheet.titleblockId = tb.id;
  sheets.push(sheet);
  notify();
  return sheet;
}

export function updateSheet(id: string, patch: Partial<SheetDef>) {
  const s = sheets.find((x) => x.id === id);
  if (!s) return;
  Object.assign(s, patch, { updatedAt: Date.now() });
  notify();
}

export function deleteSheet(id: string) {
  sheets = sheets.filter((s) => s.id !== id);
  notify();
}

export function addViewport(sheetId: string, vp: Omit<SheetViewport, "id">): SheetViewport | null {

  const viewport: SheetViewport = { ...vp, id: uid() };
  s.viewports.push(viewport);
  s.updatedAt = Date.now();

}

export function removeViewport(sheetId: string, viewportId: string) {
  const s = sheets.find((x) => x.id === sheetId);
  if (!s) return;
  s.viewports = s.viewports.filter((v) => v.id !== viewportId);

}

export function getSheetSizeMm(size: SheetDef["size"], orientation: SheetDef["orientation"]) {
  const base = SHEET_SIZES_MM[size] ?? SHEET_SIZES_MM.A1;
  if (orientation === "portrait") return { w: Math.min(base.w, base.h), h: Math.max(base.w, base.h) };
  return { w: Math.max(base.w, base.h), h: Math.min(base.w, base.h) };
}

function escapeXml(s: string) {
  return String(s)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

export function exportSheetSvg(sheetId: string, filename?: string) {

  if (!svg) return;
  const sheet = getSheet(sheetId);

  const a = document.createElement("a");

  a.download = filename ?? `${sheet?.number ?? "sheet"}.svg`;
  a.click();

}
