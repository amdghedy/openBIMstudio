## ðŸŒ Overview

**OpenBIMStudio** is a local-first, browser-based BIM platform designed to compete with heavy desktop solutions like Autodesk Revit, Navisworks, and Solibri. Built entirely on open standards, it allows AEC professionals to author, coordinate, document, and collaborate on complex building projects without installing gigabytes of software.

By leveraging **Three.js** and the **That Open Company (OBC)** ecosystem, OpenBIMStudio renders massive federated IFC models with high performance, acting as a native web-based authoring engine.

> **Status:** Early-stage / Active Development. The core authoring engine, parametrics, and IFC export are highly functional and continuously expanding.

---

## ðŸš€ Strong Sides & Core Capabilities

### 1. Revit-Class Authoring (M1 / M2)
*   **Compound Walls**: Layer-based wall generation for realistic construction (e.g., `Brick:0.1, Insulation:0.05, Gypsum:0.015`).
*   **Parametric Generators**: Create stairs (straight, L, U), railings, ramps, curtain walls, and roofs.
*   **MEP Modeling**: Native ducts, pipes, cable trays, and conduits with routing and elbow fitting generation.
*   **Intelligent Joins**: Walls, beams, and ducts auto-miter at corners using advanced geometric solvers.

### 2. Safe Parametrics & Constraints (M6)
*   **Zero-Eval Formula Engine**: A custom, secure parser (`formula-v2.ts`) that processes Revit-style formulas (e.g., `height = LevelHeight - 0.3`) without using dangerous `eval()`.
*   **Level Constraints**: Attach elements to Base/Top levels with offsets; they update automatically when levels move.
*   **Lookup Tables**: Native support for Revit-style `size_lookup()` CSV tables.

### 3. Coordination & Clash Detection
*   **Category-Based Clashes**: Run clashes between specific categories (e.g., `[Native] Beam` vs `[IFC] Pipe`) using a newly designed UI panel.
*   **Native vs IFC**: Check collisions between elements you authored in the browser and elements imported from external federated models.
*   **Triangle BVH**: High-performance geometric clash detection using bounding volume hierarchies.

### 4. Documentation & 2D Generation (M3)
*   **True Sections & Elevations**: Real geometric boolean cuts with poche fills and projected silhouettes.
*   **Smart Plan Views**: Auto-tagged doors/windows (`D1`, `W1`), room area tags, and auto-dimensioning.
*   **Dynamic Schedules**: Generate quantity takeoffs and material schedules directly from the 3D model.

### 5. Interoperability (M4)
*   **Native IFC Export**: Full IFC4 and IFC2X3 STEP file generation with valid GlobalIds and property sets.
*   **Revit Bridge**: Export/Import JSON sidecars to seamlessly exchange rooms, levels, and phases with Autodesk Revit add-ins.
*   **DWG / DXF Parsing**: Native CAD import using `libredwg-web` for 2D reference underlays.

### 6. Real-Time Collaboration (M5)
*   **Element-Level Worksharing**: Borrow and lock specific elements via a central server, preventing "last write wins" data loss.
*   **Realtime Sync**: Yjs / WebSockets integration for live cursor tracking and model synchronization.


   ```bash
   npm install
   ```


function showError(title: string, err: unknown) {
  const msg = err instanceof Error ? `${err.message}\n${err.stack ?? ""}` : String(err);
  if (isBenign(msg)) {
    console.warn(`[suppressed] ${title}: ${msg.split("\n")[0]}`);
    return;
  }
  console.error(`[OpenBIMStudio] ${title}`, err);
  errOverlay.style.display = "block";
  errOverlay.innerHTML = "";
  const pre = document.createElement("pre");
  pre.style.cssText = "margin:0;white-space:pre-wrap;font:inherit";
  pre.textContent = `âœ– ${title}\n${msg}`;
  const dismiss = document.createElement("button");
  dismiss.textContent = "Dismiss";
  dismiss.style.cssText =
    "margin-top:8px;background:#7f1d1d;color:#fecaca;border:1px solid #ef4444;" +
    "border-radius:6px;padding:4px 10px;cursor:pointer;font:12px system-ui";
  dismiss.onclick = () => { errOverlay.style.display = "none"; errOverlay.innerHTML = ""; };
  errOverlay.append(pre, dismiss);
}

window.addEventListener("error", (e) => {

window.addEventListener("unhandledrejection", (e: PromiseRejectionEvent) => {
  const reason: any = e.reason;
  const msg =
    reason instanceof Error ? reason.message
    : typeof reason === "string" ? reason
    : String(reason?.message ?? reason ?? "");
  if (isBenign(msg)) {
    console.warn("[suppressed benign] " + msg);
    e.preventDefault();
    return;
  }
  // Production: show once, do not cascade crash the app
  e.preventDefault();
  showError("Background task failed (app continues)", reason);
});

  const worlds = components.get(OBC.Worlds);
  const world = worlds.create<OBC.SimpleScene, OBC.OrthoPerspectiveCamera, OBC.SimpleRenderer>();
  world.scene = new OBC.SimpleScene(components);
  world.renderer = new OBC.SimpleRenderer(components, rightHost);
  world.camera = new OBC.OrthoPerspectiveCamera(components);

  // 3D navigation (MMB/wheel/dbl-MMB) is owned by features/navigation.ts
  // after input-router attaches. Only disable LMB/RMB/wheel on camera-controls.
  try {
    const controls: any = world.camera.controls;
    const ACTION = controls?.constructor?.ACTION || (window as any).CameraControls?.ACTION;
    const mouseButtons = controls?.mouseButtons;
    if (mouseButtons && ACTION) {
      mouseButtons.left = ACTION.NONE ?? 0;
      mouseButtons.middle = ACTION.NONE ?? 0;
      mouseButtons.right = ACTION.NONE ?? 0;
      mouseButtons.wheel = ACTION.NONE ?? 0;
    }
  } catch (error) {
    console.warn("[navigation] mouse mapping unavailable", error);
  }

  components.init();
  world.scene.setup();
  world.scene.three.background = new THREE.Color(0x0f1117);
  world.scene.three.add(new THREE.AmbientLight(0xffffff, 0.55));
  const sun = new THREE.DirectionalLight(0xffffff, 1.35);
  sun.position.set(50, 100, 40);
  world.scene.three.add(sun);
  const grids = components.get(OBC.Grids);
  const grid = grids.create(world);
  (ctx as any).grid = grid;
  (ctx as any).grids = grids;

  const groundPlane = new THREE.Mesh(
    new THREE.PlaneGeometry(1000, 1000),
    new THREE.MeshBasicMaterial({ visible: false, side: THREE.DoubleSide })
  );
  groundPlane.rotation.x = -Math.PI / 2;
  groundPlane.name = "groundPlane";
  world.scene.three.add(groundPlane);

  const fragments = components.get(OBC.FragmentsManager);
  try {
    fragments.init(await OBC.FragmentsManager.getWorker());
  } catch (e) {
    console.error("[fragments] worker init failed", e);
    throw new Error(
      "Fragments worker failed to load. Often caused by COEP/CORS or a 404 HTML page. " +
      "Restart Vite after npm install. Original: " + (e instanceof Error ? e.message : String(e))
    );
  }
  world.camera.controls.addEventListener("update", () => fragments.core.update());
  world.onCameraChanged.add((camera) => {
    for (const [, model] of fragments.list) model.useCamera(camera.three);
    fragments.core.update(true);
  });
  fragments.list.onItemSet.add(async ({ value: model }) => {
    model.useCamera(world.camera.three);
    world.scene.three.add(model.object);
    await fragments.core.update(true);
  });
  fragments.core.models.materials.list.onItemSet.add(({ value: material }) => {
    if (!("isLodMaterial" in material && (material as any).isLodMaterial)) {
      material.polygonOffset = true;
      material.polygonOffsetUnits = 1;
      material.polygonOffsetFactor = Math.random();
    }
  });

  components.get(OBC.Raycasters).get(world);

  const highlighter = components.get(OBCF.Highlighter);
  highlighter.setup({

    autoHighlightOnClick: false,
    selectMaterialDefinition: { color: new THREE.Color("#bcf124"), opacity: 1, transparent: false, renderedFaces: 0 },
  });
  highlighter.styles.set("clash", { color: new THREE.Color("#ef4444"), opacity: 0.9, transparent: true, renderedFaces: 0 });
  // Selection A = blue, Selection B = yellow (clash focus)
  highlighter.styles.set("clashA", { color: new THREE.Color("#3b82f6"), opacity: 0.95, transparent: true, renderedFaces: 0 });
  highlighter.styles.set("clashB", { color: new THREE.Color("#eab308"), opacity: 0.95, transparent: true, renderedFaces: 0 });
  // Revit-style preselection: hover feedback sits below selection priority.
  // Explicit selection style so linked-model selection is always visibly blue,
  // independent of the SDK default material definition. Keep it above preselection.
  highlighter.styles.set("select", {
    color: new THREE.Color("#2563eb"),
    opacity: 0.82,
    transparent: true,
    renderedFaces: 0,
    priority: 20,
  } as any);
  highlighter.styles.set("preselect", {
    color: new THREE.Color("#38bdf8"),
    opacity: 0.42,
    transparent: true,
    renderedFaces: 0,
    priority: 10,
  } as any);

  const [propertiesTable, updatePropertiesTable] = BUIC.tables.itemsData({ components, modelIdMap: {} });
  propertiesTable.preserveStructureOnFilter = true;
  const [modelsList] = BUIC.tables.modelsList({ components, metaDataTags: ["schema"], actions: { download: true } });

  Object.assign(ctx, {
    components, world, fragments, highlighter,
    hider: components.get(OBC.Hider),
    clipper: components.get(OBC.Clipper),
    lengthMeasurer: components.get(OBCF.LengthMeasurement),
    leftHost, rightHost, groundPlane, modelsList,
  });
  ctx.clipper.enabled = true;
  ctx.props.table = propertiesTable;
  ctx.props.update = (map: any) => updatePropertiesTable({ modelIdMap: map });

  window.addEventListener("resize", () => {
    world.renderer?.resize();
    world.camera.updateAspect();
    ctx.cam.updateClipping();
  });

  return ctx;
}

type ServiceKey = keyof ServiceMap;

const registry = new Map<ServiceKey, ServiceMap[ServiceKey]>();
const readyResolvers = new Map<ServiceKey, Array<(v: ServiceMap[ServiceKey]) => void>>();

/**
 * Register a service. Re-registration overwrites and wakes any pending
 * `ready()` calls. This is what happens during HMR or a project reload.
 */
export function register<K extends ServiceKey>(key: K, value: ServiceMap[K]): void {
  registry.set(key, value);

  // Mirror into the legacy ctx.features bag so unmigrated code keeps working.
  try {
    (ctx.features as Record<string, unknown>)[key as string] = value;
  } catch {
    /* */
  }

  const waiters = readyResolvers.get(key);
  if (waiters) {
    readyResolvers.delete(key);
    for (const cb of waiters) {
      try { cb(value); } catch { /* */ }
    }
  }
}

export function use<K extends ServiceKey>(key: K): ServiceMap[K] {
  const v = registry.get(key);
  if (!v) {
    throw new ServiceNotReadyError(key);
  }
  return v as ServiceMap[K];
}

export function tryUse<K extends ServiceKey>(key: K): ServiceMap[K] | null {
  return (registry.get(key) as ServiceMap[K]) ?? null;
}

export function hasService<K extends ServiceKey>(key: K): boolean {
  return registry.has(key);
}

/**
 * Resolve when the service is registered. If it is already registered, the
 * returned promise resolves on the next microtask so callers can always
 * `await` without branching.
 */
export function ready<K extends ServiceKey>(key: K): Promise<ServiceMap[K]> {
  const existing = registry.get(key);
  if (existing) return Promise.resolve(existing as ServiceMap[K]);

  return new Promise<ServiceMap[K]>((resolve) => {
    const list = readyResolvers.get(key) ?? [];
    list.push(resolve as (v: ServiceMap[ServiceKey]) => void);
    readyResolvers.set(key, list);
  });
}

export class ServiceNotReadyError extends Error {
  constructor(public readonly key: ServiceKey) {
    super(
      `Service "${String(key)}" was requested before it was registered. ` +
      `Call services.ready("${String(key)}") to await it, or use ` +
      `services.tryUse("${String(key)}") if it is optional.`,
    );
    this.name = "ServiceNotReadyError";
  }
}

interface PanelFeature {
  isVisible?: () => boolean;
  open?: () => void;
  close?: () => void;
  toggle?: () => void;
}

function asVisible(F: PanelFeature): boolean {
  try { return typeof F.isVisible === "function" && !!F.isVisible(); }
  catch { return false; }
}

function setVisible(F: PanelFeature, on: boolean): void {
  const now = asVisible(F);
  if (now === on) return;
  if (on && typeof F.open === "function") { F.open(); return; }
  if (!on && typeof F.close === "function") { F.close(); return; }
  if (typeof F.toggle === "function") F.toggle();
}

function reg(featureKey: string, label: string, defaultOpen = false): boolean {
  const F = (ctx.features as any)[featureKey] as PanelFeature | undefined;
  if (!F) return false;
  registerWorkspacePanel({
    key: featureKey,
    label,
    defaultOpen,
    isVisible: () => asVisible(F),
    setVisible: (on: boolean) => setVisible(F, on),
  });
  return true;
}

type ToastKind = "info" | "warn" | "success" | "error";
let toastHost: HTMLDivElement | null = null;

function ensureToastHost(): HTMLDivElement {
  if (toastHost) return toastHost;
  const h = document.createElement("div");
  h.className = "obs-toast-host";
  document.body.appendChild(h);
  toastHost = h;
  return h;
}

function toast(message: string, kind: ToastKind = "info", ms = 2400): void {
  const host = ensureToastHost();
  const el = document.createElement("div");
  el.className = "obs-toast";
  el.dataset.kind = kind;
  el.textContent = message;
  host.appendChild(el);
  setTimeout(() => {
    el.classList.add("obs-toast--out");
    setTimeout(() => el.remove(), 250);
  }, ms);
}

  ribbonMinimized = localStorage.getItem("obs.ribbon.minimized") === "1";
} catch { /* */ }

function applyRibbonMinimize(min: boolean) {
  ribbonMinimized = min;
  if (min) {
    root.classList.add("obs-ribbon--minimized");
    strip.style.display = "none";
  } else {
    root.classList.remove("obs-ribbon--minimized");
    strip.style.display = "";
  }
  try { localStorage.setItem("obs.ribbon.minimized", min ? "1" : "0"); } catch { /* */ }
  // Notify plan view / others that ribbon height changed
  window.dispatchEvent(new Event("resize"));
  window.dispatchEvent(new CustomEvent("obs-ribbon-resize"));
}

function toggleRibbonMinimize() {
  applyRibbonMinimize(!ribbonMinimized);
}

const esc = (s: string) =>
  s.replace(/[&<>"']/g, (c) =>
    ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" } as any)[c]);

interface BtnOpts {
  active?: boolean;
  title?: string;
  planned?: boolean;
  kbd?: string;
}

type Action = () => void | Promise<void>;

function b(
  icon: IconName,
  label: string,
  onclick: Action,
  activeOrOpts: boolean | BtnOpts = false,
  title = "",
): HTMLButtonElement {
  const opts: BtnOpts =
    typeof activeOrOpts === "boolean" ? { active: activeOrOpts, title } : activeOrOpts;
  const el = document.createElement("button");
  el.className = "obs-rb";
  const tip = opts.kbd ? `${label}  (${opts.kbd})` : (opts.title || label);
  el.title = tip;
  el.setAttribute("aria-label", label);
  if (opts.active) el.dataset.active = "1";
  if (opts.planned) el.dataset.planned = "1";
  el.innerHTML =
    `<span class="obs-rb__icon">${svgIcon(icon)}</span>` +
    `<span class="obs-rb__label">${esc(label)}</span>`;
  el.onclick = async () => {
    try {
      await onclick();
    } catch (err) {
      console.error("[ribbon] action failed", err);
      toast("Action failed â€” see console.", "warn", 3000);
    }
    renderStrip();
  };
  return el;
}

function toolBtn(icon: IconName, label: string, tool: string, kbd = ""): HTMLButtonElement {
  let active = false;
  try {
    // Match internal tool or ribbon preset name
    const current = getActiveTool() as string | null;
    active = current === tool || (current != null && label.length > 0 && current === tool);
    try {
      const preset = (window as any).__obsAuthoring?.getActivePreset?.();
      if (preset === tool) active = true;
    } catch { /* */ }
  } catch { /* */ }
  return b(
    icon,
    label,
    safe(() => {
      try {
        const cur = getActiveTool() as string | null;
        let preset: string | null = null;
        try { preset = (window as any).__obsAuthoring?.getActivePreset?.() ?? null; } catch { /* */ }
        if (cur === tool || preset === tool) {
          setAuthoringTool(null);
          toast(`${label} off`, "info", 1200);
        } else {
          setAuthoringTool(tool as any);
          // If Floor Plan is open, route tool into 2D plan (Revit behaviour)
          try {
            const pv = (ctx.features as any).planView;
            if (pv?.isVisible?.()) {
              pv.setTool?.(tool);
              toast(`${label} â€” draw in Floor Plan (also creates 3D)`, "success", 2000);
            } else {
              toast(`${label} â€” click in the viewport to place`, "success", 1800);
            }
          } catch {
            toast(`${label} â€” click in the viewport to place`, "success", 1800);
          }
        }
      } catch (e) {
        toast(`Tool "${label}" failed â€” see console.`, "warn");
        console.warn("[ribbon] tool failed", tool, e);
      }
    }, label),
    { active, kbd, title: `${label} modeling tool` },
  );
}

function safe(fn: Action, label = "action"): Action {
  return async () => {
    try {
      await fn();
    } catch (err) {
      console.error(`[ribbon] ${label}`, err);
      toast(`${label} failed â€” see console.`, "warn", 3000);
    }
  };
}

function renderTabs() {
  tabsRow.innerHTML = "";

  const brand = document.createElement("div");

  const version = document.createElement("span");

  version.textContent = "v54";
  brand.appendChild(version);

  for (const t of TABS) {
    // Removed tabs: BIM Tools, Analyze, Massing & Site
    if (t.id === "bim-tools" || t.id === "analyze" || t.id === "site") continue;
    const btn = document.createElement("button");
    btn.className = "obs-ribbon__tab" + (t.id === "file" ? " obs-ribbon__tab--file" : "");
    btn.textContent = t.label;
    if (t.id === activeTab) btn.dataset.active = "1";
    btn.dataset.tab = t.id;
    btn.onclick = () => {
      // Single click: if minimized, expand and switch; if already expanded, just switch tab
      if (ribbonMinimized) {
        applyRibbonMinimize(false);
      }
      activeTab = t.id;
      renderTabs();
      renderStrip();
    };
    // Double-click on tab â†’ toggle minimize (Revit behavior: shrink ribbon so only tabs show)

      toggleRibbonMinimize();

function renderStripImpl() {
  strip.innerHTML = "";

  const tab = TABS.find((t) => t.id === activeTab) ?? TABS[0];
  let items: HTMLElement[] = [];
  try {
    items = tab.items();
  } catch (err) {
    console.error(`[ribbon] items() threw for tab "${tab.id}":`, err);
    strip.innerHTML =
      `<div style="color:#fca5a5;padding:20px;font:12px/1.5 monospace;white-space:pre-wrap">` +
      `âš  Tab "${tab.id}" threw during build:\n` +
      `${err instanceof Error ? err.message : String(err)}\n\n` +
      `Open the browser console (F12) for the full trace.</div>`;
    return;
  }
  if (!Array.isArray(items)) return;

  const groups: { label: string; items: HTMLElement[] }[] = [];
  const hints: HTMLElement[] = [];
  let cur: { label: string; items: HTMLElement[] } = { label: "", items: [] };

  for (const it of items) {
    if (!it || !(it instanceof HTMLElement)) continue;
    if (it.dataset.obsSep === "1") {
      if (cur.items.length) groups.push(cur);
      cur = { label: "", items: [] };
      continue;
    }
    if (it.dataset.obsGroup) {
      cur.label = it.dataset.obsGroup;
      continue;
    }
    if (it.dataset.obsHint === "1") {
      hints.push(it);
      continue;
    }
    cur.items.push(it);
  }
  if (cur.items.length) groups.push(cur);

  if (!groups.length && !hints.length) {
    const fallback = document.createElement("div");
    fallback.style.cssText =
      "padding:20px;color:var(--text-tertiary,#6b7280);font:12px system-ui";
    fallback.textContent = `Tab "${tab.label}" has no commands yet.`;
    strip.appendChild(fallback);
    return;
  }

  for (const g of groups) {
    const wrap = document.createElement("div");
    wrap.className = "obs-ribbon-group";
    const bodyEl = document.createElement("div");
    bodyEl.className = "obs-ribbon-group__body";
    for (const el of g.items) bodyEl.appendChild(el);
    wrap.appendChild(bodyEl);
    if (g.label) {
      const lbl = document.createElement("div");
      lbl.className = "obs-ribbon-group__label";
      lbl.textContent = g.label;
      wrap.appendChild(lbl);
    }
    strip.appendChild(wrap);
  }

  for (const h of hints) {
    h.className = "obs-ribbon-hint";
    strip.appendChild(h);
  }
}

function setupLeftDock(mainRow: HTMLElement) {
  const leftWidthKey = "obs.leftPanel.width";
  const dockKey = "obs.leftPanel.dock";
  const minW = 320;
  const maxW = 700;

  mainRow.style.position = "relative";

  let width = 400;
  try {
    width = Math.max(
      minW,
      Math.min(maxW, Number(localStorage.getItem(leftWidthKey)) || 400),
    );
  } catch { /* */ }
  ctx.leftHost.style.width = `${width}px`;
  ctx.leftHost.style.minWidth = `${minW}px`;
  ctx.leftHost.style.maxWidth = `${maxW}px`;
  ctx.leftHost.style.flex = "0 0 auto";
  ctx.leftHost.style.position = "relative";

  const handle = document.createElement("div");
  handle.style.cssText =
    "position:absolute;top:0;bottom:0;right:-3px;width:6px;cursor:ew-resize;" +
    "background:transparent;z-index:20";
  handle.title = "Drag to resize Â· Double-click to reset";
  ctx.leftHost.appendChild(handle);

  let resizing = false;
  let startX = 0;
  let startW = 0;

  handle.addEventListener("pointerdown", (e) => {
    resizing = true;
    startX = e.clientX;
    startW = ctx.leftHost.getBoundingClientRect().width;
    (e.target as Element).setPointerCapture(e.pointerId);
    document.body.style.cursor = "ew-resize";
    e.preventDefault();
  });
  handle.addEventListener("pointermove", (e) => {
    if (!resizing) return;
    const dx = e.clientX - startX;
    const w = Math.max(minW, Math.min(maxW, startW + dx));
    ctx.leftHost.style.width = `${w}px`;
    window.dispatchEvent(new Event("resize"));
  });
  handle.addEventListener("pointerup", (e) => {
    if (!resizing) return;
    resizing = false;
    (e.target as Element).releasePointerCapture(e.pointerId);
    document.body.style.cursor = "";
    try {
      localStorage.setItem(
        leftWidthKey,
        String(Math.round(ctx.leftHost.getBoundingClientRect().width)),
      );
    } catch { /* */ }
  });
  handle.addEventListener("dblclick", () => {
    ctx.leftHost.style.width = "400px";
    window.dispatchEvent(new Event("resize"));
    try { localStorage.setItem(leftWidthKey, "400"); } catch { /* */ }
  });

  function applyDock(side: "left" | "right") {
    if (side === "right") {
      mainRow.style.flexDirection = "row-reverse";
      ctx.leftHost.style.borderRight = "none";
      ctx.leftHost.style.borderLeft = "1px solid var(--border-default)";
    } else {
      mainRow.style.flexDirection = "row";
      ctx.leftHost.style.borderLeft = "none";
      ctx.leftHost.style.borderRight = "1px solid var(--border-default)";
    }
    window.dispatchEvent(new Event("resize"));
  }
  try {
    const dock = (localStorage.getItem(dockKey) as "left" | "right") || "left";
    applyDock(dock);
  } catch { /* */ }

  (ctx.features as any).leftPanel = {
    setWidth: (w: number) => {
      const clamped = Math.max(minW, Math.min(maxW, w));
      ctx.leftHost.style.width = `${clamped}px`;
      window.dispatchEvent(new Event("resize"));
      try { localStorage.setItem(leftWidthKey, String(Math.round(clamped))); } catch { /* */ }
    },
    getWidth: () => ctx.leftHost.getBoundingClientRect().width,
    dockLeft: () => {
      applyDock("left");
      try { localStorage.setItem(dockKey, "left"); } catch { /* */ }
    },
    dockRight: () => {
      applyDock("right");
      try { localStorage.setItem(dockKey, "right"); } catch { /* */ }
    },
    toggle: () => toggleLeftPanel(),
  };
}

    const message = String(d.message ?? "");
    const timeout = d.timeout ?? d.ms ?? (kind === "error" ? 6000 : 2400);
    toast(message, kind, timeout);
