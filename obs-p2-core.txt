/**
 * Single authoritative pointer / wheel / Escape router for the 3D viewport.
 *
 * Key properties:
 *   â€¢ Gesture ownership â€” the first handler to accept a `drag-start` owns
 *     that gesture. Only the owner sees `drag-move` and `drag-end`.
 *   â€¢ Priority ordering â€” higher priority runs first. Gizmo (100) beats
 *     box-select (50) beats context-menu (20) beats authoring (10).
 *   â€¢ Cancel registration â€” Escape handlers run in reverse registration
 *     order; the first to return true consumes the key.
 *   â€¢ Camera auto-pause â€” the router pauses CameraControls for the duration
 *     of any claimed drag and releases it on drag-end.
 *   â€¢ Hover â€” throttled to one dispatch per frame; all subscribers receive it.
 *
 * The router attaches capture-phase listeners to the host so it sees events
 * before anything else, but only calls preventDefault / stopPropagation when
 * a handler claims the gesture. Unclaimed input falls through to the camera
 * controls and any legacy listeners still present.
 */
import { ctx } from "./context";
import { pauseCameraControls, forceResumeCameraControls } from "./camera-gate";

export type PointerButton = 0 | 1 | 2;

export interface Mods {
  ctrl: boolean;
  shift: boolean;
  alt: boolean;
  meta: boolean;
}

export type Gesture =
  | { kind: "click"; button: PointerButton; mods: Mods; event: PointerEvent }
  | { kind: "dblclick"; button: PointerButton; mods: Mods; event: PointerEvent }
  | { kind: "drag-start"; button: PointerButton; mods: Mods; event: PointerEvent }
  | { kind: "drag-move"; button: PointerButton; mods: Mods; dx: number; dy: number; event: PointerEvent }
  | { kind: "drag-end"; button: PointerButton; mods: Mods; dx: number; dy: number; moved: boolean; event: PointerEvent }
  | { kind: "rmb-click"; mods: Mods; event: PointerEvent }
  | { kind: "wheel"; event: WheelEvent; mods: Mods };

export type GestureHandler = (g: Gesture) => boolean;
export type HoverHandler = (e: PointerEvent) => void;
export type CancelHandler = () => boolean;


/** True when the event originated on a floating panel / form control â€” not the canvas. */
function isUiEventTarget(t: EventTarget | null): boolean {
  const el = t as HTMLElement | null;
  if (!el || !el.closest) return false;
  if (el.closest(
    "input,textarea,select,button,label,a," +
    "[contenteditable=true],[data-obs-window-controls]",
  )) return true;
  // Any dockable / floating panel shell
  if (el.closest(
    '#obs-ai-chat,#obs-bim-toolbox,[id^="obs-"],[data-obs-panel],.obs-panel,.obs-toast-host',
  )) {
    // Still allow canvas-like surfaces inside panels (e.g. plan view)
    if (el.closest("canvas")) return false;
    return true;
  }
  // Generic: if it's not the WebGL canvas, treat as UI when inside rightHost children that are not the canvas parent
  const tag = el.tagName;
  if (tag === "CANVAS") return false;
  return false;
}

const DRAG_THRESHOLD = 4;
const DBLCLICK_MS = 320;
const DBLCLICK_PX = 6;

interface Sub { fn: GestureHandler; priority: number; order: number; }

class InputRouterImpl {
  private host: HTMLElement | null = null;
  private subs: Sub[] = [];
  private hoverHandlers: HoverHandler[] = [];
  private cancelHandlers: CancelHandler[] = [];
  private orderCounter = 0;

  private down: {
    x: number; y: number;
    button: PointerButton;
    mods: Mods;
    moved: boolean;
    id: number;
  } | null = null;

  private owner: GestureHandler | null = null;
  private releaseCamera: (() => void) | null = null;

  private lastClick: { t: number; x: number; y: number; button: PointerButton } | null = null;

  private hoverPending = false;
  private lastHoverEvent: PointerEvent | null = null;

  /* â”€â”€ Lifecycle â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */
  attach(host: HTMLElement): void {
    if (this.host === host) return;
    this.detach();
    this.host = host;
    host.addEventListener("pointerdown", this.onDown, { capture: true });
    host.addEventListener("pointermove", this.onMove, { capture: true });
    host.addEventListener("pointerup", this.onUp, { capture: true });
    host.addEventListener("pointercancel", this.onUp, { capture: true });
    host.addEventListener("wheel", this.onWheel, { capture: true, passive: false });
    host.addEventListener("contextmenu", this.onContextMenu, { capture: true });
    window.addEventListener("keydown", this.onKeyDown, { capture: true });
  }

  detach(): void {
    const host = this.host;
    if (host) {
      host.removeEventListener("pointerdown", this.onDown, true);
      host.removeEventListener("pointermove", this.onMove, true);
      host.removeEventListener("pointerup", this.onUp, true);
      host.removeEventListener("pointercancel", this.onUp, true);
      host.removeEventListener("wheel", this.onWheel, true);
      host.removeEventListener("contextmenu", this.onContextMenu, true);
    }
    window.removeEventListener("keydown", this.onKeyDown, true);
    this.host = null;
    this.down = null;
    this.releaseOwner();
  }

  /* â”€â”€ Subscription â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */
  subscribe(h: GestureHandler, priority = 0): () => void {
    const sub: Sub = { fn: h, priority, order: this.orderCounter++ };
    this.subs.push(sub);
    this.subs.sort((a, b) => b.priority - a.priority || a.order - b.order);
    return () => {
      const i = this.subs.indexOf(sub);
      if (i >= 0) this.subs.splice(i, 1);
    };
  }

  subscribeHover(h: HoverHandler): () => void {
    this.hoverHandlers.push(h);
    return () => {
      const i = this.hoverHandlers.indexOf(h);
      if (i >= 0) this.hoverHandlers.splice(i, 1);
    };
  }

  registerCancel(h: CancelHandler): () => void {
    this.cancelHandlers.push(h);
    return () => {
      const i = this.cancelHandlers.indexOf(h);
      if (i >= 0) this.cancelHandlers.splice(i, 1);
    };
  }

  hasOwner(): boolean { return this.owner !== null; }

  /* â”€â”€ Internals â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */
  private mods(e: MouseEvent | WheelEvent | PointerEvent): Mods {
    return { ctrl: e.ctrlKey, shift: e.shiftKey, alt: e.altKey, meta: e.metaKey };
  }

  private dispatch(g: Gesture): GestureHandler | null {
    for (const s of this.subs) {
      try {
        if (s.fn(g)) return s.fn;
      } catch (err) {
        console.error("[input-router] handler threw", err);
      }
    }
    return null;
  }

  private releaseOwner(): void {
    this.owner = null;
    if (this.releaseCamera) {
      try { this.releaseCamera(); } catch { /* */ }
      this.releaseCamera = null;
    }
  }

  private scheduleHover(e: PointerEvent): void {
    this.lastHoverEvent = e;
    if (this.hoverPending) return;
    this.hoverPending = true;
    requestAnimationFrame(() => {
      this.hoverPending = false;
      const ev = this.lastHoverEvent;
      if (!ev) return;
      for (const h of this.hoverHandlers) {
        try { h(ev); } catch (err) { console.error("[input-router] hover threw", err); }
      }
    });
  }

  /* â”€â”€ Pointer events â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€ */
  private onDown = (e: PointerEvent): void => {
    if (e.button !== 0 && e.button !== 1 && e.button !== 2) return;
    if (isUiEventTarget(e.target)) return;
    if (this.down && this.down.id !== e.pointerId) return;
    this.down = {
      x: e.clientX, y: e.clientY,
      button: e.button as PointerButton,
      mods: this.mods(e),
      moved: false,
      id: e.pointerId,
    };
    this.owner = null;
  };

  private onMove = (e: PointerEvent): void => {
    if (!this.down) { this.scheduleHover(e); return; }
    if (this.down.id !== e.pointerId) return;

    const dx = e.clientX - this.down.x;
    const dy = e.clientY - this.down.y;
    const dist = Math.hypot(dx, dy);

    if (!this.down.moved && dist >= DRAG_THRESHOLD) {
      this.down.moved = true;
      const owner = this.dispatch({
        kind: "drag-start",
        button: this.down.button,
        mods: this.mods(e),
        event: e,
      });
      if (owner) {
        this.owner = owner;
        this.releaseCamera = pauseCameraControls();
        e.preventDefault();
        e.stopPropagation();
      }
    }

    if (this.owner) {
      try {
        this.owner({
          kind: "drag-move",
          button: this.down.button,
          mods: this.mods(e),
          dx, dy,
          event: e,
        });
      } catch (err) {
        console.error("[input-router] owner drag-move threw", err);
      }
      e.preventDefault();
      e.stopPropagation();
    }
  };

  private onUp = (e: PointerEvent): void => {
    if (!this.down || this.down.id !== e.pointerId) return;
    const d = this.down;
    this.down = null;

    const dx = e.clientX - d.x;
    const dy = e.clientY - d.y;
    const moved = d.moved || Math.hypot(dx, dy) >= DRAG_THRESHOLD;

    if (this.owner) {
      try {
        this.owner({
          kind: "drag-end",
          button: d.button,
          mods: this.mods(e),
          dx, dy,
          moved: true,
          event: e,
        });
      } catch (err) {
        console.error("[input-router] owner drag-end threw", err);
      }
      e.preventDefault();
      e.stopPropagation();
      this.releaseOwner();
      return;
    }

    if (moved) {
      // Drag completed but nothing claimed it â€” reset double-click timing.
      this.lastClick = null;
      return;
    }

    if (d.button === 2) {
      if (this.dispatch({ kind: "rmb-click", mods: this.mods(e), event: e })) {
        e.preventDefault();
        e.stopPropagation();
      }
      this.lastClick = null;
      return;
    }

    const now = performance.now();
    const isDbl = !!this.lastClick
      && now - this.lastClick.t < DBLCLICK_MS
      && Math.hypot(e.clientX - this.lastClick.x, e.clientY - this.lastClick.y) < DBLCLICK_PX
      && this.lastClick.button === d.button;

    const g: Gesture = isDbl
      ? { kind: "dblclick", button: d.button, mods: this.mods(e), event: e }
      : { kind: "click", button: d.button, mods: this.mods(e), event: e };

    if (this.dispatch(g)) {
      e.preventDefault();
      e.stopPropagation();
    }
    this.lastClick = { t: now, x: e.clientX, y: e.clientY, button: d.button };
  };

  private onWheel = (e: WheelEvent): void => {
    if (isUiEventTarget(e.target)) return;
    if (this.dispatch({ kind: "wheel", event: e, mods: this.mods(e) })) {
      e.preventDefault();
      e.stopPropagation();
    }
  };

  private onContextMenu = (e: MouseEvent): void => {
    // The application owns the menu. Never let the browser default leak
    // through over the viewport.
    e.preventDefault();
  };

  private onKeyDown = (e: KeyboardEvent): void => {
    if (e.key !== "Escape") return;
    for (let i = this.cancelHandlers.length - 1; i >= 0; i--) {
      try {
        if (this.cancelHandlers[i]()) {
          e.preventDefault();
          e.stopImmediatePropagation();
          // Always clear a stuck camera pause after cancel.
          forceResumeCameraControls();
          this.releaseOwner();
          return;
        }
      } catch (err) {
        console.error("[input-router] cancel handler threw", err);
      }
    }
    // Escape with no cancel handler: still unstick camera if a drag left it paused.
    forceResumeCameraControls();
    this.releaseOwner();
  };
}

export const inputRouter = new InputRouterImpl();

export function setupInputRouter(): () => void {
  if (!ctx.rightHost) {
    console.warn("[input-router] rightHost not ready");
    return () => {};
  }
  inputRouter.attach(ctx.rightHost);
  (ctx.features as any).inputRouter = inputRouter;
  console.log("[input-router] attached");
  return () => inputRouter.detach();
}

/**
 * Options Registry â€” central place for app-level preferences.
 */
import { ctx } from "./context";
import { diagnostics } from "./diagnostics";

export type OptionType = "boolean" | "number" | "string" | "select";

export interface OptionBase {
  id: string;
  group: string;
  label: string;
  description?: string;
  type: OptionType;
  default: boolean | number | string;
  apply?: (value: any) => void;
}

export interface BooleanOption extends OptionBase {
  type: "boolean";
  default: boolean;
}
export interface NumberOption extends OptionBase {
  type: "number";
  default: number;
  min?: number;
  max?: number;
  step?: number;
  unit?: string;
}
export interface StringOption extends OptionBase {
  type: "string";
  default: string;
  placeholder?: string;
}
export interface SelectOption extends OptionBase {
  type: "select";
  default: string;
  choices: Array<{ value: string; label: string }>;
}

export type OptionDef = BooleanOption | NumberOption | StringOption | SelectOption;

const STORAGE_KEY = "obs.options.v1";
const registry = new Map<string, OptionDef>();
const values = new Map<string, any>();
const listeners = new Set<(id: string, value: any) => void>();

function loadPersisted(): Record<string, any> {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return {};
    const parsed = JSON.parse(raw);
    return parsed && typeof parsed === "object" ? parsed : {};
  } catch {
    return {};
  }
}

function persist(): void {
  try {
    const out: Record<string, any> = {};
    for (const [id, v] of values) out[id] = v;
    localStorage.setItem(STORAGE_KEY, JSON.stringify(out));
  } catch (err) {
    diagnostics.warn("options.persist", "could not save options", err);
  }
}

function validate(def: OptionDef, v: any): boolean {
  switch (def.type) {
    case "boolean":
      return typeof v === "boolean";
    case "number":
      return typeof v === "number" && Number.isFinite(v);
    case "string":
      return typeof v === "string";
    case "select":
      return def.choices.some((c) => c.value === v);
  }
}

export function registerOption(def: OptionDef): void {
  if (registry.has(def.id)) {
    diagnostics.warn("options.register", `duplicate option id "${def.id}"`);
    return;
  }
  registry.set(def.id, def);
}

export function listOptions(group?: string): OptionDef[] {
  const all = [...registry.values()];
  return group ? all.filter((o) => o.group === group) : all;
}

export function getOption<T = any>(id: string): T {
  if (values.has(id)) return values.get(id) as T;
  const def = registry.get(id);
  if (!def) return undefined as any;
  const persisted = loadPersisted();
  const v =
    id in persisted && validate(def, persisted[id]) ? persisted[id] : def.default;
  values.set(id, v);
  return v as T;
}

export function setOption(id: string, value: any): void {
  const def = registry.get(id);
  if (!def) {
    diagnostics.warn("options.set", `unknown option "${id}"`);
    return;
  }
  if (!validate(def, value)) {
    diagnostics.warn("options.set", `invalid value for "${id}"`, value);
    return;
  }
  values.set(id, value);
  try {
    def.apply?.(value);
  } catch (err) {
    diagnostics.warn(`options.apply.${id}`, "apply threw", err);
  }
  persist();
  for (const l of listeners) {
    try {
      l(id, value);
    } catch {
      /* */
    }
  }
}

export function onOptionChanged(cb: (id: string, value: any) => void): () => void {
  listeners.add(cb);
  return () => listeners.delete(cb);
}

export function applyAllOptions(): void {
  for (const def of registry.values()) {
    const v = getOption(def.id);
    try {
      def.apply?.(v);
    } catch (err) {
      diagnostics.warn(`options.apply.${def.id}`, "apply threw", err);
    }
  }
}

function renderOptions(group?: string): void {
  if (!dialog) return;
  const sidebar = dialog.querySelector('[data-role="sidebar"]') as HTMLElement;
  const content = dialog.querySelector('[data-role="content"]') as HTMLElement;
  if (!sidebar || !content) return;

  const groups = [...new Set(listOptions().map((o) => o.group))].sort();
  if (!currentGroup || (group && groups.includes(group))) {
    currentGroup = group ?? groups[0] ?? null;
  }

  sidebar.innerHTML = "";
  for (const g of groups) {
    const b = document.createElement("button");
    b.textContent = g;
    const active = g === currentGroup;
    b.style.cssText =

    b.onclick = () => {
      currentGroup = g;
      renderOptions();
    };
    sidebar.appendChild(b);

    content.innerHTML =
      `<div style="color:#6b7280;text-align:center;padding:40px">No options registered.</div>`;

/**
 * Locale-aware translation service.
 *
 * Design goals:
 *   - Zero dependencies â€” works with plain JSON catalogs
 *   - Namespaced keys via dot notation: t("clash.status.selectModels")
 *   - Plural rules via Intl.PluralRules â€” no ICU parser needed
 *   - Variable interpolation with {name} syntax
 *   - RTL detection and Direction state
 *   - Feature/plugin catalogs can register at runtime
 *   - Locale change fires an event so UI can re-render
 *
 * Catalog shape:
 *   {
 *     "meta": { "name": "English", "nativeName": "English", "dir": "ltr" },
 *     "clash": {
 *       "status": {
 *         "selectModels": "Select models in A and B, then Run Test.",
 *         "found": { "one": "1 clash found.", "other": "{count} clashes found." }
 *       }
 *     }
 *   }
 */
import { diagnostics } from "./diagnostics";

export type Locale =
  | "en" | "ar" | "de" | "fr" | "es" | "pt-BR" | "it"
  | "zh-CN" | "zh-TW" | "ja" | "ko"
  | "ru" | "tr" | "pl" | "nl"
  | "hi" | "he" | "fa" | "ur";

export type TextDirection = "ltr" | "rtl";

const RTL_LOCALES: ReadonlySet<Locale> = new Set<Locale>([
  "ar", "he", "fa", "ur",
]);

export interface CatalogMeta {
  name: string;
  nativeName: string;
  dir: TextDirection;
}

export type CatalogLeaf = string | PluralForms;
export type PluralForms = Partial<Record<
  Intl.LDMLPluralRule | "other",
  string
>>;
export type CatalogNode = CatalogLeaf | { [key: string]: CatalogNode };

/* -------------------------------------------------------------------- */

const STORAGE_KEY = "obs.locale";
const FALLBACK_LOCALE: Locale = "en";
const MISSING_MARKER = "\u2400";

type ChangeListener = (locale: Locale) => void;

class I18nService {
  private catalogs = new Map<Locale, Catalog>();
  private current: Locale = FALLBACK_LOCALE;
  private listeners = new Set<ChangeListener>();
  private pluralRulesCache = new Map<Locale, Intl.PluralRules>();
  private missingKeys = new Set<string>();
  private booted = false;

  /* -- Registration --------------------------------------------------- */

  register(locale: Locale, catalog: Partial<Catalog>): void {
    const existing = this.catalogs.get(locale);
    if (!existing) {
      this.catalogs.set(locale, catalog as Catalog);
      return;
    }
    const merged = deepMerge(existing, catalog);
    this.catalogs.set(locale, merged as Catalog);
  }

  registerAll(bundle: Partial<Record<Locale, Partial<Catalog>>>): void {
    for (const [locale, cat] of Object.entries(bundle)) {
      if (!cat) continue;
      this.register(locale as Locale, cat);
    }
  }

  /* -- Lifecycle ------------------------------------------------------ */

  boot(): void {
    if (this.booted) return;
    this.booted = true;

    let saved: Locale | null = null;
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (raw) saved = raw as Locale;
    } catch { /* */ }

    const detected = saved ?? this.detectBrowserLocale();
    this.setLocale(detected, { silent: !saved });
  }

  private detectBrowserLocale(): Locale {
    const langs = navigator.languages?.length
      ? navigator.languages
      : [navigator.language];
    for (const l of langs) {
      const exact = l as Locale;
      if (this.catalogs.has(exact)) return exact;

      // Keep the raw string so we can compare against values that are
      // intentionally not in the Locale union (e.g. "zh").
      const baseStr = l.split("-")[0];
      const base = baseStr as Locale;
      if (this.catalogs.has(base)) return base;

      // Special case for Chinese: "zh" â†’ "zh-CN"
      if (baseStr === "zh") return "zh-CN";
    }
    return FALLBACK_LOCALE;
  }

  getLocale(): Locale {
    return this.current;
  }

  getDirection(): TextDirection {
    const cat = this.catalogs.get(this.current);
    return cat?.meta?.dir ?? (RTL_LOCALES.has(this.current) ? "rtl" : "ltr");
  }

  isRTL(): boolean {
    return this.getDirection() === "rtl";
  }

  listLocales(): Array<{ id: Locale; meta: CatalogMeta }> {
    const out: Array<{ id: Locale; meta: CatalogMeta }> = [];
    for (const [id, cat] of this.catalogs) {
      out.push({ id, meta: cat.meta });
    }
    out.sort((a, b) => a.meta.name.localeCompare(b.meta.name));
    return out;
  }

  setLocale(locale: Locale, opts: { silent?: boolean } = {}): void {
    if (!this.catalogs.has(locale)) {
      diagnostics.warn(
        "i18n.setLocale",
        `locale "${locale}" not registered â€” falling back to en`,
      );
      locale = FALLBACK_LOCALE;
    }
    if (this.current === locale && this.booted) return;
    this.current = locale;

    try {
      localStorage.setItem(STORAGE_KEY, locale);
    } catch { /* */ }

    this.applyDocumentAttributes();

    if (!opts.silent) {
      for (const cb of this.listeners) {
        try {
          cb(locale);
        } catch (err) {
          diagnostics.warn("i18n.listener", "listener threw", err);
        }
      }
    }
  }

  private applyDocumentAttributes(): void {
    if (typeof document === "undefined") return;
    const dir = this.getDirection();
    document.documentElement.lang = this.current;
    document.documentElement.dir = dir;
    document.body?.setAttribute("data-locale", this.current);
    document.body?.setAttribute("data-dir", dir);
  }

  onChange(cb: ChangeListener): () => void {
    this.listeners.add(cb);
    return () => this.listeners.delete(cb);
  }

  /* -- Translation ---------------------------------------------------- */

  t(key: string, vars?: Record<string, string | number>): string {
    const value =
      this.lookup(this.current, key) ??
      this.lookup(FALLBACK_LOCALE, key);

    if (value === undefined) {
      if (!this.missingKeys.has(key)) {
        this.missingKeys.add(key);
        diagnostics.warn(
          "i18n.missing",
          `missing key "${key}" for locale "${this.current}"`,
        );
      }
      return key;
    }

    if (typeof value === "string") {
      return interpolate(value, vars);
    }

    const count = vars?.count;
    const n = typeof count === "number" ? count : Number(count ?? 0);
    const rule = this.pluralRule(this.current, n);
    const chosen =
      value[rule] ??
      value.other ??
      value.one ??
      Object.values(value)[0] ??
      key;
    return interpolate(chosen, vars);
  }

  has(key: string): boolean {
    return (
      this.lookup(this.current, key) !== undefined ||
      this.lookup(FALLBACK_LOCALE, key) !== undefined
    );
  }

  clearMissing(): void {
    this.missingKeys.clear();
  }

  listMissing(): string[] {
    return [...this.missingKeys].sort();
  }

  private pluralRule(locale: Locale, count: number): Intl.LDMLPluralRule {
    let rules = this.pluralRulesCache.get(locale);
    if (!rules) {
      try {
        rules = new Intl.PluralRules(locale);
      } catch {
        rules = new Intl.PluralRules(FALLBACK_LOCALE);
      }
      this.pluralRulesCache.set(locale, rules);
    }
    return rules.select(count);
  }

  private lookup(locale: Locale, key: string): CatalogLeaf | undefined {
    const cat = this.catalogs.get(locale);
    if (!cat) return undefined;
    const parts = key.split(".");
    let node: CatalogNode | undefined = cat as unknown as CatalogNode;
    for (const p of parts) {
      if (node === undefined || typeof node === "string") return undefined;
      if (isPluralForms(node)) return undefined;
      node = node[p];
    }
    if (node === undefined) return undefined;
    if (typeof node === "string" || isPluralForms(node)) return node;
    return undefined;
  }
}

function isPluralForms(v: unknown): v is PluralForms {
  if (!v || typeof v !== "object") return false;
  const keys = Object.keys(v);
  if (!keys.length) return false;
  return keys.every((k) =>
    ["zero", "one", "two", "few", "many", "other"].includes(k),
  );
}

function interpolate(
  template: string,
  vars?: Record<string, string | number>,
): string {
  if (!vars) return template;
  return template.replace(/\{(\w+)\}/g, (match, name) => {
    const v = vars[name];
    if (v === undefined) return match;
    return String(v);
  });
}

function deepMerge<T extends Record<string, unknown>>(
  a: T,
  b: Partial<T>,
): T {
  const out: Record<string, unknown> = { ...a };
  for (const [k, v] of Object.entries(b)) {
    const prev = out[k];
    if (
      v && typeof v === "object" && !Array.isArray(v) &&
      prev && typeof prev === "object" && !Array.isArray(prev)
    ) {
      out[k] = deepMerge(
        prev as Record<string, unknown>,
        v as Record<string, unknown>,
      );
    } else {
      out[k] = v;
    }
  }
  return out as T;
}

/**
 * Revit-style feet-inches. Precision is the inch fraction denominator:
 *   fractionDenominator = 16 â†’ "5'-6 1/2""
 *   showUnit = false       â†’ omit the trailing double-quote
 */
function formatFeetInches(
  meters: number,
  opts: { precision: number; showUnit: boolean; fractions: boolean; denominator: number },
): string {
  if (!Number.isFinite(meters)) return "â€”";
  const negative = meters < 0;
  const totalInches = Math.abs(meters) / LENGTH_TO_M.in;
  const sign = negative ? "-" : "";

  let feet = Math.floor(totalInches / 12);
  let inches = totalInches - feet * 12;

  if (opts.fractions) {
    const denom = Math.max(2, Math.min(64, opts.denominator));
    let numerator = Math.round((inches - Math.floor(inches)) * denom);
    let whole = Math.floor(inches);
    if (numerator === denom) { whole += 1; numerator = 0; }
    if (whole === 12) { feet += 1; whole = 0; }

    let inchStr = String(whole);
    if (numerator > 0) {
      const g = gcd(numerator, denom);
      inchStr += ` ${numerator / g}/${denom / g}`;
    }

    const feetStr = `${feet}'`;
    if (opts.showUnit) return `${sign}${feetStr}-${inchStr}"`;
    return `${sign}${feetStr}-${inchStr}`;
  }

  const inchesRounded = Math.round(inches * 100) / 100;
  const inchStr = formatNumberLocale(inchesRounded, opts.precision);
  const feetStr = `${feet}'`;
  return opts.showUnit ? `${sign}${feetStr}-${inchStr}"` : `${sign}${feetStr}-${inchStr}`;
}

function gcd(a: number, b: number): number {
  return b === 0 ? a : gcd(b, a % b);
}

function parseFeetInches(input: string): number | null {
  // e.g. 5'-6 1/2"  |  5' 6"  |  5'
  const s = input.trim().replace(/[\u2019\u02BC]/g, "'");
  const ftIn = s.match(/^(-?\d+(?:[.,]\d+)?)\s*'\s*(?:-\s*)?(.*)$/);
  if (!ftIn) return null;
  const feet = parseNumberLocale(ftIn[1].replace(",", "."));
  if (feet == null) return null;

  const rest = ftIn[2].replace(/["â€³]/g, "").trim();
  if (!rest) return feet * LENGTH_TO_M.ft;

  // Possible fraction: "6 1/2" or "6-1/2"
  const frac = rest.match(/^(\d+(?:[.,]\d+)?)(?:\s+|[-])(\d+)\s*\/\s*(\d+)$/);
  if (frac) {
    const whole = parseNumberLocale(frac[1].replace(",", ".")) ?? 0;
    const num = parseFloat(frac[2]);
    const den = parseFloat(frac[3]);
    const inches = whole + (den ? num / den : 0);
    return feet * LENGTH_TO_M.ft + inches * LENGTH_TO_M.in;
  }
  const plain = parseNumberLocale(rest.replace(",", "."));
  if (plain == null) return feet * LENGTH_TO_M.ft;
  return feet * LENGTH_TO_M.ft + plain * LENGTH_TO_M.in;
}

/** Locale-aware number parse: accepts "1.234,56", "1,234.56", "1 234,56". */
function parseNumberLocale(s: string): number | null {
  if (!s) return null;
  // Heuristic: if there's a single comma with digits after, treat as decimal.
  // If both, the last one wins.
  const lastComma = s.lastIndexOf(",");
  const lastDot = s.lastIndexOf(".");
  let normalized: string;

  if (lastComma === -1 && lastDot === -1) normalized = s;
  else if (lastComma === -1) normalized = s;
  else if (lastDot === -1) normalized = s.replace(",", ".");
  else if (lastComma > lastDot) normalized = s.replace(/\./g, "").replace(",", ".");
  else normalized = s.replace(/,/g, "");

  const n = Number(normalized);
  return Number.isFinite(n) ? n : null;
}

/**
 * Uniform 3D spatial hash for broadphase candidate lookup.
 *
 * Cell size is chosen from the median item diagonal. Each item is inserted
 * into every cell its AABB overlaps. Queries return the union of items in
 * the same cells â€” a superset of the true AABB-overlap set, so the caller
 * must still do the actual Box3 test (which is cheap).
 */
import * as THREE from "three";

export class SpatialHash {
  private readonly cellSize: number;
  private readonly buckets = new Map<string, number[]>();
  private readonly itemBoxes: THREE.Box3[] = [];

  constructor(cellSize: number) {
    this.cellSize = Math.max(cellSize, 1e-3);
  }

  get size(): number {
    return this.itemBoxes.length;
  }

  get cell(): number {
    return this.cellSize;
  }

  /** Insert an item's world-space AABB and get its index back. */
  insert(box: THREE.Box3): number {
    const idx = this.itemBoxes.length;
    this.itemBoxes.push(box.clone());

    const i0 = Math.floor(box.min.x / this.cellSize);
    const i1 = Math.floor(box.max.x / this.cellSize);
    const j0 = Math.floor(box.min.y / this.cellSize);
    const j1 = Math.floor(box.max.y / this.cellSize);
    const k0 = Math.floor(box.min.z / this.cellSize);
    const k1 = Math.floor(box.max.z / this.cellSize);

    const cells = (i1 - i0 + 1) * (j1 - j0 + 1) * (k1 - k0 + 1);
    if (!Number.isFinite(cells) || cells > 4096) {
      const arr = this.buckets.get("*") ?? [];
      arr.push(idx);
      this.buckets.set("*", arr);
      return idx;
    }

    for (let i = i0; i <= i1; i++) {
      for (let j = j0; j <= j1; j++) {
        for (let k = k0; k <= k1; k++) {
          const key = `${i},${j},${k}`;
          const arr = this.buckets.get(key);
          if (arr) arr.push(idx);
          else this.buckets.set(key, [idx]);
        }
      }
    }
    return idx;
  }

  /**
   * Fill `out` with candidate indices whose cells overlap the query box.
   * `out` is cleared first. Excludes `selfIdx` if it appears.
   */
  query(box: THREE.Box3, out: Set<number>, selfIdx = -1): void {
    out.clear();

    const i0 = Math.floor(box.min.x / this.cellSize);
    const i1 = Math.floor(box.max.x / this.cellSize);
    const j0 = Math.floor(box.min.y / this.cellSize);
    const j1 = Math.floor(box.max.y / this.cellSize);
    const k0 = Math.floor(box.min.z / this.cellSize);
    const k1 = Math.floor(box.max.z / this.cellSize);

    const cells = (i1 - i0 + 1) * (j1 - j0 + 1) * (k1 - k0 + 1);
    if (!Number.isFinite(cells) || cells > 4096) {
      for (let i = 0; i < this.itemBoxes.length; i++) {
        if (i !== selfIdx) out.add(i);
      }
      return;
    }

    for (let i = i0; i <= i1; i++) {
      for (let j = j0; j <= j1; j++) {
        for (let k = k0; k <= k1; k++) {
          const arr = this.buckets.get(`${i},${j},${k}`);
          if (!arr) continue;
          for (const idx of arr) {
            if (idx !== selfIdx) out.add(idx);
          }
        }
      }
    }

    const global = this.buckets.get("*");
    if (global) {
      for (const idx of global) {
        if (idx !== selfIdx) out.add(idx);
      }
    }
  }

  boxAt(idx: number): THREE.Box3 | undefined {
    return this.itemBoxes[idx];
  }

  /** Choose a sensible cell size from a sample of item sizes. */
  static suggestCellSize(boxes: readonly THREE.Box3[], fallback = 2): number {
    if (boxes.length === 0) return fallback;
    const sample = boxes.slice(0, Math.min(boxes.length, 512));
    const diag = new THREE.Vector3();
    const sizes: number[] = [];
    for (const b of sample) {
      b.getSize(diag);
      const d = Math.hypot(diag.x, diag.y, diag.z);
      if (Number.isFinite(d) && d > 0) sizes.push(d);
    }
    if (!sizes.length) return fallback;
    sizes.sort((a, b) => a - b);
    const median = sizes[Math.floor(sizes.length / 2)];
    return Math.max(0.25, Math.min(median * 1.5, 25));
  }
}

/**
 * Camera-controls gating.
 *
 * Different @thatopen/components versions expose the underlying
 * CameraControls instance at different paths. This module probes once,
 * caches the reference, and provides a reference-counted pause so multiple
 * features (gizmo, box-select, authoring grip drag) can hold the camera
 * paused simultaneously without one re-enabling it while another still
 * needs it disabled.
 */
import { ctx } from "./context";

let cached: any = null;
let cacheAttempted = false;

function findControls(): any {
  if (cacheAttempted) return cached;
  cacheAttempted = true;
  try {
    const cam: any = (ctx.world as any)?.camera;
    if (!cam) return (cached = null);

    // Prefer an object that exposes `.enabled` directly (the actual controls).
    if (cam.controls && typeof cam.controls === "object") {
      if (typeof cam.controls.enabled === "boolean") return (cached = cam.controls);
      if (cam.controls.controls && typeof cam.controls.controls.enabled === "boolean") {
        return (cached = cam.controls.controls);
      }
    }
    if (cam.cameraControls && typeof cam.cameraControls.enabled === "boolean") {
      return (cached = cam.cameraControls);
    }
  } catch {
    /* swallow â€” a missing controls object is a valid state during boot */
  }
  return (cached = null);
}

export function resetCameraGateCache(): void {
  cached = null;
  cacheAttempted = false;
  holders = 0;
  lastEnabled = null;
}

let holders = 0;
let lastEnabled: boolean | null = null;

function applyEnabled(on: boolean): void {
  if (lastEnabled === on) return;
  const c = findControls();
  if (c) c.enabled = on;
  lastEnabled = on;
}

/**
 * Pause camera interaction. Returns a release function. Callers must call
 * the release exactly once. Nested calls are supported via refcount.
 */
export function pauseCameraControls(): () => void {
  holders++;
  applyEnabled(false);
  let released = false;
  return () => {
    if (released) return;
    released = true;
    holders = Math.max(0, holders - 1);
    if (holders === 0) applyEnabled(true);
  };
}

export function cameraControlsPaused(): boolean {
  return holders > 0;
}

/** Force-reset the gate â€” used by recovery code after a stuck gesture. */
export function forceResumeCameraControls(): void {
  holders = 0;
  applyEnabled(true);
}

 *   - normalizes deltaY (deltaMode + clamp) and applies Revit convention
 *     (wheel up = zoom in). setInvert(true) flips it for reversed devices.
 *   - Shift = x3, plus the registered nav.wheelZoomMultiplier option.
 *   - ignores wheel while __OBS_MMB_HELD (no zoom during pan).
 *   - ignores wheel over panels / ribbon / property palette / plan SVG /
 *     text inputs, so 2D plan zoom and UI scrolling are not stolen.
 *   - stopImmediatePropagation() so no secondary module double-zooms.
 *   - sets window.__OBS_WHEEL_AUTHORITY = true permanently after install,
 *     which navigation.ts reads to no-op its router wheel branch (this is
 *     order-independent: the flag is set once at boot, not per-event).
 *
 * It does NOT touch CameraControls mouse mapping (navigation owns that).
 */

let installed = false;
let enabled = true;
let invert = false;
let multiplier = 1;

const UI_BLOCK =
  ".obs-panel,[data-obs-panel],#obs-ai-chat,#obs-bim-toolbox,.obs-ribbon," +
  ".obs-property,.obs-dialog,.obs-adv-toolbar,.obs-plan-view,#plan-svg," +
  "input,textarea,select,button,label,a,[contenteditable=true]";

function w(): Any { return window as Any; }

function readMultiplier(): number {
  try {
    const reg: Any = (w().__obsOptionsRegistry ?? null);
    const v = reg?.get?.("nav.wheelZoomMultiplier");
    if (typeof v === "number" && v > 0) return v;
  } catch { /* registry optional */ }
  return multiplier;
}

function normalizeDelta(e: WheelEvent): number {
  let d = e.deltaY;
  if (e.deltaMode === 1) d *= 16;        // DOM_DELTA_LINE
  else if (e.deltaMode === 2) d *= 100;  // DOM_DELTA_PAGE
  // clamp trackpad momentum spikes
  if (d > 600) d = 600;
  if (d < -600) d = -600;
  return d;
}

function isOverViewport(target: EventTarget | null): boolean {
  const el = target as Any;
  if (!el) return false;
  if (typeof el.closest === "function" && el.closest(UI_BLOCK)) return false;
  // must be inside the 3D host / canvas, not a side panel
  const c = ctx as Any;
  const host: Any = c.rightHost;
  if (host && host.contains && !host.contains(el)) {
    // allow the canvas itself even if rightHost ref is stale
    const tag = String(el.tagName ?? "").toUpperCase();
    if (tag !== "CANVAS") return false;
  }
  return true;
}

function applyZoom(ndelta: number, shift: boolean): void {
  const c = ctx as Any;
  const cam: Any = c.world?.camera?.three;
  const controls: Any = c.world?.camera?.controls;
  if (!cam) return;

  const speed = 0.0016 * readMultiplier() * (shift ? 3 : 1);
  // Revit: wheel up (ndelta < 0) => zoom in => factor > 1
  const signed = invert ? -ndelta : ndelta;
  const factor = Math.exp(-signed * speed);
  if (!Number.isFinite(factor) || factor <= 0) return;

  const isOrtho = cam.isOrthographicCamera === true;

  if (isOrtho) {
    cam.zoom = Math.max(1e-4, cam.zoom * factor);
    cam.updateProjectionMatrix();
  } else {
    const target = new THREE.Vector3();
    let got = false;
    try { if (typeof controls?.getTarget === "function") { controls.getTarget(target); got = true; } } catch { /* */ }
    if (!got) {
      const fwd = new THREE.Vector3();
      cam.getWorldDirection(fwd);
      target.copy(cam.position).addScaledVector(fwd, 10);
    }
    const dir = new THREE.Vector3().subVectors(cam.position, target);
    const dist = dir.length();
    if (dist < 1e-6) return;
    const newDist = Math.max(1e-4, dist / factor);
    dir.normalize().multiplyScalar(newDist);
    const np = new THREE.Vector3().addVectors(target, dir);
    try {
      controls.setLookAt(np.x, np.y, np.z, target.x, target.y, target.z, false);
      controls.update?.(0);
    } catch {
      cam.position.copy(np);
    }
  }

  try { window.dispatchEvent(new CustomEvent("obs:camera-changed")); } catch { /* */ }
}

function onWheel(e: WheelEvent): void {
  if (!enabled) return;
  if ((w() as Any).__OBS_MMB_HELD) return;          // panning: no zoom
  if (!isOverViewport(e.target)) return;            // panels / UI / plan svg
  const nd = normalizeDelta(e);
  if (nd === 0) return;
  e.preventDefault();
  e.stopImmediatePropagation();
  applyZoom(nd, !!e.shiftKey);
}

export function setWheelInvert(next: boolean): boolean {
  invert = !!next;
  return invert;
}
export function setWheelMultiplier(next: number): number {
  multiplier = Number.isFinite(next) && next > 0 ? next : 1;
  return multiplier;
}
export function setWheelEnabled(next: boolean): boolean {
  enabled = !!next;
  return enabled;
}
export function isWheelEnabled(): boolean { return enabled; }

export function setupWheelArbiter(): void {
  if (installed) return;
  installed = true;

  // Permanent flag read by navigation.ts router-wheel branch (order-independent).
  (w() as Any).__OBS_WHEEL_AUTHORITY = true;

  window.addEventListener("wheel", onWheel, { capture: true, passive: false });

  (w() as Any).__obsWheelFix = {
    setInvert: setWheelInvert,
    setMultiplier: setWheelMultiplier,
    setEnabled: setWheelEnabled,
    isEnabled: isWheelEnabled,
    getState: () => ({ enabled, invert, multiplier: readMultiplier() }),
  };

  console.log(
    "[wheel-arbiter] single owner ready: Revit direction (up=zoom-in), " +
    "deltaMode normalized, Shift=x3, ignored while MMB held / over panels. " +
    "Flip direction with __obsWheelFix.setInvert(true) if your device is reversed.",
  );
}

import { ctx } from "./context";

type Any = Record<string, any>;

/*
 * Input arbiter.
 *
 * This module does NOT install its own pan/orbit/zoom behaviour and does NOT
 * mutate CameraControls mouse buttons (navigation.ts already forces those OFF
 * and is the single driver). Instead it maintains a few WINDOW-LEVEL FLAGS that
 * navigation.ts and the rest of the app read, plus a couple of capture-phase
 * guards that are safe regardless of listener registration order:
 *
 *   __OBS_MMB_HELD   : true while middle button is down
 *   __OBS_MMB_MOVED  : true once the MMB drag exceeded a few pixels
 *   __OBS_MMB_DOWN_T : timestamp of MMB down (watchdog)
 *   __OBS_BLOCK_MMB_CLICK : swallow the synthetic click after a dragged MMB
 *
 * Why flags instead of stopPropagation? Because navigation.ts / selection.ts
 * register their window capture listeners during app boot, i.e. BEFORE this
 * arbiter is imported by critical-gaps-boot. A later capture listener cannot
 * cancel an earlier one. Flags, however, are simply *read* by the earlier
 * handlers at the moment they run, so coordination works in any order.
 *
 * The only things we actively suppress at capture phase are:
 *   - wheel events while __OBS_MMB_HELD (no zoom-out during a pan)
 *   - the click that the browser synthesises after a DRAGGED MMB (so a pan
 *     never also selects / focuses / fits)
 * Both are safe to suppress because they are never legitimate during those
 * states under the documented Revit map.
 */

let installed = false;

const MMB_MOVE_THRESHOLD = 3;     // px before a press becomes a drag
const MMB_CLICK_WINDOW = 380;     // ms for a genuine double-MMB (kept for fit)
const WATCHDOG_MS = 8000;         // clear stuck HELD if no up arrived

function w(): Any { return window as Any; }

function clearMMB(): void {
  const W = w();
  W.__OBS_MMB_HELD = false;
  W.__OBS_MMB_MOVED = false;
  W.__OBS_MMB_DOWN_T = 0;
}

function isTextInput(target: EventTarget | null): boolean {
  const el = target as Any;
  if (!el) return false;
  const tag = String(el.tagName ?? "").toLowerCase();
  if (tag === "input" || tag === "textarea" || tag === "select") return true;
  if (el.isContentEditable) return true;
  if (typeof el.closest === "function" &&
      el.closest('input, textarea, select, [contenteditable="true"]')) return true;
  return false;
}

function activeAuthoringTool(): string | null {
  const c = ctx as Any;
  const a = (c.features as Any)?.authoring as Any;
  const t =
    a?.getActiveTool?.() ?? a?.activeTool ?? a?.currentTool ?? null;
  if (typeof t === "string" && t && t !== "select") return t;
  return null;
}

function cancelActiveOperation(): boolean {
  let canceled = false;
  const W = w();

  // 1) drafting overlay (left-mouse tool)
  const d = W.__obsDrafting2D;
  if (d && typeof d.getTool === "function" && d.getTool()) {
    d.setTool(null);
    canceled = true;
  }

  // 2) authoring placement tool
  const c = ctx as Any;
  const a = (c.features as Any)?.authoring as Any;
  if (activeAuthoringTool()) {
    if (typeof a?.setAuthoringTool === "function") {
      a.setAuthoringTool(null);
      canceled = true;
    }
  }

  // 3) measure / section / walk (best effort through known globals)
  const m = W.__obsMeasure;
  if (m && (typeof m.isEnabled === "function" ? m.isEnabled() : m.enabled)) {
    m.setMeasureEnabled?.(false); m.disable?.(); canceled = true;
  }
  const s = W.__obsSection;
  if (s && (typeof s.isSectionOn === "function" ? s.isSectionOn() : s.enabled)) {
    s.setSectionOn?.(false); s.disable?.(); canceled = true;
  }
  const wk = W.__obsWalkMode;
  if (wk && (typeof wk.isActive === "function" ? wk.isActive() : wk.enabled)) {
    wk.disable?.(); wk.setActive?.(false); canceled = true;
  }

  // 4) always drop MMB arbitration state on a cancel
  clearMMB();

  if (canceled) {
    try { window.dispatchEvent(new CustomEvent("obs:operation-cancelled")); } catch { /* */ }
  }
  return canceled;
}

function installArbiter(): void {
  const W = w();

  // ---- pointer tracking (capture; we never stopPropagation on down/move so
  //      navigation.ts still receives the gesture and can pan/orbit) ----
  window.addEventListener("pointerdown", (e: PointerEvent) => {
    if (e.button !== 1) return;            // middle only
    W.__OBS_MMB_HELD = true;
    W.__OBS_MMB_MOVED = false;
    W.__OBS_MMB_DOWN_T = performance.now();
    W.__OBS_MMB_START_X = e.clientX;
    W.__OBS_MMB_START_Y = e.clientY;
  }, true);

  window.addEventListener("pointermove", (e: PointerEvent) => {
    if (!W.__OBS_MMB_HELD) return;
    const dx = e.clientX - (W.__OBS_MMB_START_X ?? e.clientX);
    const dy = e.clientY - (W.__OBS_MMB_START_Y ?? e.clientY);
    if (Math.hypot(dx, dy) > MMB_MOVE_THRESHOLD) {
      W.__OBS_MMB_MOVED = true;
    }
  }, true);

  const onUp = (e: PointerEvent | MouseEvent) => {
    const be = e as PointerEvent;
    if (typeof be.button === "number" && be.button !== 1) return;
    const moved = !!W.__OBS_MMB_MOVED;
    const downT = Number(W.__OBS_MMB_DOWN_T || 0);
    clearMMB();
    // A dragged MMB must not also produce a select / focus / fit click.
    if (moved) {
      W.__OBS_BLOCK_MMB_CLICK = true;
      window.setTimeout(() => { W.__OBS_BLOCK_MMB_CLICK = false; }, 0);
    }
    // remember a clean (no-move) MMB click time for genuine double-MMB fit
    if (!moved && downT > 0) {
      W.__OBS_LAST_MMB_CLICK_T = performance.now();
    }
  };
  window.addEventListener("pointerup", onUp as any, true);
  window.addEventListener("pointercancel", onUp as any, true);
  window.addEventListener("mouseup", onUp as any, true);   // backup
  window.addEventListener("blur", () => clearMMB(), true);
  document.addEventListener("visibilitychange", () => {
    if (document.hidden) clearMMB();
  });

  // ---- wheel guard: no zoom while panning (capture; safe to suppress) ----
  window.addEventListener("wheel", (e: WheelEvent) => {
    if (W.__OBS_MMB_HELD) {
      e.preventDefault();
      e.stopPropagation();
    }
  }, { capture: true, passive: false });

  // ---- swallow the synthetic click after a dragged MMB ----
  window.addEventListener("click", (e: MouseEvent) => {
    if (e.button === 1 && W.__OBS_BLOCK_MMB_CLICK) {
      e.preventDefault();
      e.stopPropagation();
      W.__OBS_BLOCK_MMB_CLICK = false;
    }
  }, true);

  // ---- watchdog: never leave HELD stuck if an up was missed ----
  window.setInterval(() => {
    if (W.__OBS_MMB_HELD && W.__OBS_MMB_DOWN_T &&
        performance.now() - Number(W.__OBS_MMB_DOWN_T) > WATCHDOG_MS) {
      clearMMB();
    }
  }, 1000);

  // ---- keyboard / right-click cancel (Revit behaviour) ----
  window.addEventListener("keydown", (e: KeyboardEvent) => {
    if (isTextInput(e.target)) return;
    if (e.key === "Escape") {
      if (cancelActiveOperation()) {
        e.preventDefault();
        e.stopPropagation();
      }
    }
  }, true);

  window.addEventListener("contextmenu", (e: MouseEvent) => {
    // Only cancel when an operation is actually active; otherwise let
    // navigation.ts open its right-click context menu untouched.
    if (cancelActiveOperation()) {
      e.preventDefault();
      e.stopPropagation();
    }
  }, true);

  W.__obsRevitMousePolicy = {
    cancelOperation: cancelActiveOperation,
    isMMBHeld: () => !!w().__OBS_MMB_HELD,
    isMMBMoved: () => !!w().__OBS_MMB_MOVED,
    clearMMB,
  };
}

export function setupRevitMousePolicy(): void {
  if (installed) return;
  installed = true;

  // NOTE: we deliberately do NOT call any disableNativeCameraMouse() here.
  // navigation.ts already forces CameraControls mouse/touch OFF and re-asserts
  // it via forceResumeCameraControls(); a second writer raced with it and was a
  // source of the rotate/zoom-on-release artefact.

  installArbiter();

  console.log(
    "[revit-mouse-policy] arbiter ready: flags-only coordination; " +
    "wheel blocked while MMB held; dragged-MMB click suppressed; " +
    "Esc / RMB cancel active tool; navigation.ts remains sole MMB/wheel owner",
  );
}

export type DiagLevel = "info" | "warn" | "error";

export interface DiagEvent {
  id: number;
  ts: number;
  level: DiagLevel;
  scope: string;
  message: string;
  error?: unknown;
}

const MAX_EVENTS = 500;
const listeners = new Set<(e: DiagEvent) => void>();

let counter = 0;
let events: DiagEvent[] = [];

function record(
  level: DiagLevel,
  scope: string,
  message: unknown,
  error?: unknown,
): DiagEvent {
  const ev: DiagEvent = {
    id: ++counter,
    ts: Date.now(),
    level,
    scope,
    message: typeof message === "string" ? message : String(message ?? ""),
    error,
  };
  events.push(ev);
  if (events.length > MAX_EVENTS) events.shift();

  const fn =
    level === "error" ? console.error : level === "warn" ? console.warn : console.info;
  if (error !== undefined) fn(`[${scope}]`, message, error);
  else fn(`[${scope}]`, message);

  for (const l of listeners) {
    try {
      l(ev);
    } catch {
      /* */
    }
  }
  refreshBadge();
  return ev;
}

export const diagnostics = {
  info: (scope: string, message: unknown) => record("info", scope, message),
  warn: (scope: string, message: unknown, error?: unknown) =>
    record("warn", scope, message, error),
  error: (scope: string, message: unknown, error?: unknown) =>
    record("error", scope, message, error),
  list: (): readonly DiagEvent[] => events,
  clear: () => {
    events = [];
    refreshBadge();
  },
  counts: (): { info: number; warn: number; error: number } => {
    const out = { info: 0, warn: 0, error: 0 };
    for (const e of events) out[e.level]++;
    return out;
  },
  onEvent: (cb: (e: DiagEvent) => void): (() => void) => {
    listeners.add(cb);
    return () => listeners.delete(cb);
  },
  installGlobalHandlers: () => {
    // main.ts already shows an overlay; we only ring-buffer here.
    window.addEventListener("error", (ev) => {
      record("error", "window.error", ev.message, ev.error);
    });
    window.addEventListener("unhandledrejection", (ev) => {
      const reason = (ev as PromiseRejectionEvent).reason ?? "unhandled rejection";
      record("error", "window.unhandledrejection", reason);
    });
  },
};

  return BENIGN.some((re) => re.test(msg));
}

export function installGlobalErrorHandlers(): void {
  window.addEventListener("error", (e) => {
    const msg = String(e.error?.message ?? e.message ?? "");
    if (isBenign(msg)) {
      e.preventDefault();
      return;
    }
    logError(e.error ?? e.message, "window.error");
  });
  window.addEventListener("unhandledrejection", (e) => {
    const reason = e.reason;
    const msg =
      reason instanceof Error ? reason.message
      : typeof reason === "string" ? reason
      : String((reason as { message?: string })?.message ?? reason ?? "");
    if (isBenign(msg)) {
      console.warn("[suppressed benign] " + msg);
      e.preventDefault();
      return;
    }
    logError(e.reason, "unhandled-rejection");
  });
}

export async function guarded<T>(
  fn: () => Promise<T> | T,
  context: string,
  kind: NotifyKind = "error",
): Promise<T | null> {
  try {
    return await fn();
  } catch (err) {
    const entry = logError(err, context);
    notify(kind, `${context}: ${entry.message}`);
    return null;
  }
}

/**
 * Bounded-concurrency async map.
 *
 * Runs at most `concurrency` tasks in parallel. Preserves input order in the
 * result array. Aborts early if the signal fires, rejecting with an
 * AbortError so the caller can distinguish a cancelled run from a failure.
 */
export class AbortError extends Error {
  constructor(message = "Aborted") {
    super(message);
    this.name = "AbortError";
  }
}

export interface PMapOptions {
  signal?: AbortSignal;
  onProgress?: (done: number, total: number) => void;
  /** Stop scheduling new work after this many failures. Default: reject fast. */
  continueOnError?: boolean;
}

export async function pMap<T, R>(
  items: readonly T[],
  concurrency: number,
  fn: (item: T, index: number) => Promise<R>,
  opts: PMapOptions = {},
): Promise<R[]> {
  const total = items.length;
  if (total === 0) return [];

  const limit = Math.max(1, Math.min(concurrency, total));
  const out: R[] = new Array(total);
  let cursor = 0;
  let done = 0;
  let firstError: unknown = null;

  const runWorker = async (): Promise<void> => {
    for (;;) {
      if (opts.signal?.aborted) throw new AbortError();
      if (firstError && !opts.continueOnError) throw firstError;

      const i = cursor++;
      if (i >= total) return;

      try {
        out[i] = await fn(items[i], i);
      } catch (err) {
        if (firstError === null) firstError = err;
        if (!opts.continueOnError) throw err;
      } finally {
        done++;
        opts.onProgress?.(done, total);
      }
    }
  };

  await Promise.all(Array.from({ length: limit }, () => runWorker()));
  if (firstError && !opts.continueOnError) throw firstError;
  return out;
}
