import http from "node:http";
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const __dirname = path.dirname(fileURLToPath(import.meta.url));

const dataDir = path.join(__dirname, "data");
const projectsFile = path.join(dataDir, "projects.json");
const roomsFile = path.join(dataDir, "rooms.json");

const PORT = Number(process.env.PORT || 4000);
const MAX_BODY = 200 * 1024 * 1024;
const PRESENCE_TIMEOUT_MS = 45000;
const MAX_CHANGES_PER_ROOM = 500;

const sseClients = new Map();

function ensureStore() {
  fs.mkdirSync(dataDir, { recursive: true });

  if (!fs.existsSync(projectsFile)) {
    fs.writeFileSync(projectsFile, "{}", "utf8");
  }

  if (!fs.existsSync(roomsFile)) {
    fs.writeFileSync(roomsFile, "{}", "utf8");
  }
}

function loadJson(file, fallback) {
  ensureStore();

  try {
    return JSON.parse(fs.readFileSync(file, "utf8"));
  } catch {
    return fallback;
  }
}

function saveJson(file, value) {
  ensureStore();
  fs.writeFileSync(file, JSON.stringify(value, null, 2), "utf8");
}

let projects = loadJson(projectsFile, {});
let rooms = loadJson(roomsFile, {});

function send(res, status, body) {
  const data =
    typeof body === "string"
      ? body
      : JSON.stringify(body ?? {}, null, 2);

  res.writeHead(status, {
    "Content-Type": "application/json; charset=utf-8",
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Headers": "Content-Type",
    "Access-Control-Allow-Methods": "GET,POST,PUT,OPTIONS",
    "Content-Length": Buffer.byteLength(data),
  });

  res.end(data);
}

function readBody(req) {
  return new Promise((resolve, reject) => {
    let size = 0;
    const chunks = [];

    req.on("data", (chunk) => {
      size += chunk.length;

      if (size > MAX_BODY) {
        reject(new Error("Payload too large"));
        req.destroy();
        return;
      }

      chunks.push(chunk);
    });

    req.on("end", () => {
      resolve(Buffer.concat(chunks).toString("utf8"));
    });

    req.on("error", reject);
  });
}

function getRoom(roomId) {
  if (!rooms[roomId]) {
    rooms[roomId] = {
      id: roomId,
      clients: {},
      changes: [],
      updatedAt: new Date().toISOString(),
    };
  }

  return rooms[roomId];
}

function touchRoom(room) {
  room.updatedAt = new Date().toISOString();
  saveJson(roomsFile, rooms);
}

function presenceList(roomId) {
  const room = getRoom(roomId);
  const now = Date.now();

  const list = [];

  for (const c of Object.values(room.clients)) {
    if (!c || !c.clientId) continue;
    if (now - Number(c.updatedAt || 0) > PRESENCE_TIMEOUT_MS) continue;

    list.push({
      clientId: c.clientId,
      name: c.name || "Guest",
      cursor: c.cursor || null,
      updatedAt: c.updatedAt,
    });
  }

  return list;
}

function broadcast(roomId, event, payload) {
  const set = sseClients.get(roomId);

  if (!set || !set.size) {
    return;
  }

  const msg = `event: ${event}\ndata: ${JSON.stringify(payload)}\n\n`;

  for (const res of set) {
    try {
      res.write(msg);
    } catch {
      try {
        res.end();
      } catch {
        // ignore
      }
    }
  }
}

function addSse(roomId, res) {
  if (!sseClients.has(roomId)) {
    sseClients.set(roomId, new Set());
  }

  sseClients.get(roomId).add(res);

  res.on("close", () => {
    const set = sseClients.get(roomId);

    if (set) {
      set.delete(res);

      if (!set.size) {
        sseClients.delete(roomId);
      }
    }
  });
}

function addChange(roomId, change) {
  const room = getRoom(roomId);

  const item = {
    id: change.id || `chg-${Date.now()}-${Math.random().toString(16).slice(2)}`,
    clientId: change.clientId,
    type: change.type || "unknown",
    payload: change.payload ?? {},
    at: new Date().toISOString(),
  };

  room.changes.push(item);

  if (room.changes.length > MAX_CHANGES_PER_ROOM) {
    room.changes = room.changes.slice(-MAX_CHANGES_PER_ROOM);
  }

  touchRoom(room);
  broadcast(roomId, "change", item);

  return item;
}

setInterval(() => {
  let changed = false;
  const now = Date.now();

  for (const roomId of Object.keys(rooms)) {
    const room = rooms[roomId];

    if (!room || !room.clients) continue;

    for (const clientId of Object.keys(room.clients)) {
      const c = room.clients[clientId];

      if (!c || now - Number(c.updatedAt || 0) > PRESENCE_TIMEOUT_MS) {
        delete room.clients[clientId];
        changed = true;
      }
    }

    if (changed) {
      broadcast(roomId, "presence", presenceList(roomId));
    }
  }

  if (changed) {
    saveJson(roomsFile, rooms);
  }
}, 15000);

const server = http.createServer(async (req, res) => {
  if (req.method === "OPTIONS") {
    return send(res, 204, "");
  }

  const url = new URL(
    req.url || "/",
    `http://${req.headers.host || "localhost"}`,
  );

  const projectMatch = url.pathname.match(/^\/api\/projects\/([^/]+)$/);

  if (projectMatch) {
    const id = decodeURIComponent(projectMatch[1]);

    if (req.method === "GET") {
      const project = projects[id];

      if (!project) {
        return send(res, 404, { error: "Project not found" });
      }

      return send(res, 200, project);
    }

    if (req.method === "PUT") {
      try {
        const raw = await readBody(req);
        const body = raw ? JSON.parse(raw) : {};

        const project = {
          id,
          name: body.name || id,
          data: body.data ?? body,
          updatedAt: new Date().toISOString(),
        };

        projects[id] = project;
        saveJson(projectsFile, projects);

        return send(res, 200, project);
      } catch (e) {
        return send(res, 400, {
          error: e instanceof Error ? e.message : String(e),
        });
      }
    }

    return send(res, 405, { error: "Method not allowed" });
  }

  const roomEventsMatch = url.pathname.match(/^\/api\/rooms\/([^/]+)\/events$/);

  if (roomEventsMatch && req.method === "GET") {
    const roomId = decodeURIComponent(roomEventsMatch[1]);
    getRoom(roomId);

    res.writeHead(200, {
      "Content-Type": "text/event-stream; charset=utf-8",
      "Cache-Control": "no-cache, no-transform",
      Connection: "keep-alive",
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Headers": "Content-Type",
    });

    res.write("retry: 2000\n\n");
    res.write(`event: hello\ndata: ${JSON.stringify({ roomId, at: Date.now() })}\n\n`);
    res.write(`event: presence\ndata: ${JSON.stringify(presenceList(roomId))}\n\n`);

    addSse(roomId, res);
    return;
  }

  const roomBaseMatch = url.pathname.match(/^\/api\/rooms\/([^/]+)$/);

  if (roomBaseMatch && req.method === "GET") {
    const roomId = decodeURIComponent(roomBaseMatch[1]);
    const room = getRoom(roomId);

    return send(res, 200, {
      id: roomId,
      presence: presenceList(roomId),
      changes: room.changes.slice(-80),
      updatedAt: room.updatedAt,
    });
  }

  const roomActionMatch = url.pathname.match(
    /^\/api\/rooms\/([^/]+)\/(join|heartbeat|leave|change)$/,
  );

  if (roomActionMatch && req.method === "POST") {
    const roomId = decodeURIComponent(roomActionMatch[1]);
    const action = roomActionMatch[2];

    let body = {};

    try {
      const raw = await readBody(req);
      body = raw ? JSON.parse(raw) : {};
    } catch {
      return send(res, 400, { error: "Invalid JSON" });
    }

    const room = getRoom(roomId);
    const clientId = String(body.clientId || "").trim();

    if (!clientId) {
      return send(res, 400, { error: "clientId required" });
    }

    if (action === "join" || action === "heartbeat") {
      room.clients[clientId] = {
        clientId,
        name: String(body.name || "Guest"),
        cursor: body.cursor ?? null,
        updatedAt: Date.now(),
      };

      touchRoom(room);
      broadcast(roomId, "presence", presenceList(roomId));

      return send(res, 200, { ok: true, presence: presenceList(roomId) });
    }

    if (action === "leave") {
      delete room.clients[clientId];
      touchRoom(room);
      broadcast(roomId, "presence", presenceList(roomId));

      return send(res, 200, { ok: true });
    }

    if (action === "change") {
      const change = addChange(roomId, {
        id: body.id,
        clientId,
        type: body.type,
        payload: body.payload,
      });

      return send(res, 200, { ok: true, change });
    }
  }

  return send(res, 404, {
    error:
      "Not found. Use /api/projects/:id or /api/rooms/:room or /api/rooms/:room/events or /api/rooms/:room/join",
  });
});

server.listen(PORT, () => {

  console.log(`[collab] Rooms API:      /api/rooms/:room`);
  console.log(`[collab] Live SSE API:   /api/rooms/:room/events`);
});

function splitArgs(s) {
  const out = []; let depth = 0, cur = "", inStr = false;
  for (let i = 0; i < s.length; i++) {
    const c = s[i];
    if (inStr) { cur += c; if (c === "'") { if (s[i + 1] === "'") { cur += "'"; i++; } else inStr = false; } continue; }
    if (c === "'") { inStr = true; cur += c; continue; }
    if (c === "(") depth++; if (c === ")") depth--;
    if (c === "," && depth === 0) { out.push(cur); cur = ""; } else cur += c;
  }
  out.push(cur); return out;
}
function parse(text) {
  const data = text.split("DATA;\n")[1].split("\nENDSEC;")[0].split("\n").filter(Boolean);
  const ents = new Map();
  for (const line of data) {
    const m = /^#(\d+)=([A-Z0-9_]+)\((.*)\);$/.exec(line);
    assert.ok(m, "bad line: " + line.slice(0, 100));
    assert.ok(!ents.has(+m[1]), "duplicate id " + m[1]);
    ents.set(+m[1], { name: m[2], args: m[3] === "" ? [] : splitArgs(m[3]), raw: m[3] });
  }
  return ents;
}

function validate(text, schema) {
  const ents = parse(text);
  assert.ok(text.includes(`FILE_SCHEMA(('${schema}'));`));
  const guids = new Set();
  for (const [id, e] of ents) {
    // refs resolve
    for (const m of e.raw.matchAll(/#(\d+)/g)) assert.ok(ents.has(+m[1]), `#${id} ${e.name} -> missing #${m[1]}`);
    // reals contain a decimal point (REAL attributes of the geometry entities)
    if (e.name === "IFCCARTESIANPOINT" || e.name === "IFCDIRECTION") for (const n of e.args[0].slice(1, -1).split(",")) assert.ok(/\./.test(n), `integer in real list: ${e.raw}`);
    if (e.name === "IFCEXTRUDEDAREASOLID") assert.ok(/\./.test(e.args[3]), "depth must be REAL: " + e.args[3]);
    const want = COUNTS[schema][e.name];
    if (want !== undefined) assert.strictEqual(e.args.length, want, `${schema} ${e.name} #${id} has ${e.args.length} attrs, want ${want}`);
    // rooted entities carry a valid, unique GlobalId
    if (/^IFC(WALL|WALLSTANDARDCASE|SLAB|COLUMN|BEAM|MEMBER|FOOTING|COVERING|DOOR|WINDOW|OPENINGELEMENT|DUCTSEGMENT|PIPESEGMENT|CABLECARRIERSEGMENT|FLOWSEGMENT|BUILDINGELEMENTPROXY|SPACE|BUILDINGSTOREY|BUILDING|SITE|PROJECT|REL[A-Z]+|PROPERTYSET|ELEMENTQUANTITY)$/.test(e.name)) {

## Using it

| Element | Start it | Interaction |
|---|---|---|
| Stairs (straight / L / U) | existing Stairs tool or shortcut, ribbon button, palette: "Stairs: straight run / L-shape / U-shape" | click start, click end of the first flight (Shift = orthogonal) |

# OpenBIMStudio - Revit gaps, Milestone 3 (part 1): drawings that show the model

M3 is the documentation milestone. Before building anything I traced the real drawing pipeline (views -> scene -> sheet SVG /
PDF). Findings, then what this package changes.

## What I found

| Area | Finding (verified in the code) |
|---|---|
| Plans | The generator had no rooms at all (the `room` text kind existed but nothing produced it) and no poche. Stairs, ramps and curtain walls fell into a generic "thin box" branch, **railings were drawn as closed loops**, roofs and ceilings as plain outlines. No door / window marks. |
| Elevations | Every element was its bounding rectangle: no door / window openings, no hidden surfaces (everything drawn, nothing occluded), no roof slope, no glazing. |
| Sections | Walls were cut as if always perpendicular to the cut line (thickness used as width), no openings, stairs and roofs as rectangles. |
| Sheet PDF | Polygon fills use pdf-lib `drawSvgPath(d, {x:0, y:0})` with page coordinates that are already y-up; `drawSvgPath` is y-down, so those fills very probably fall below the page (poche, tag bubbles, glazing would be missing in printed sheets). I could not run pdf-lib here, so treat this as "probable, from reading the code". |
| View templates | `ViewDef.templateId` exists but nothing reads it. There is no per-view Visibility/Graphics for 2D views. |
| Two systems | The old plan panel (`plan-view-panel.ts`, 156 KB) has its own drawing code. Sheets and PDF use the newer Scene2D pipeline (`modules/views`). M3 changes the Scene2D pipeline; the old panel is untouched. |

## What changed

**Plans** (`m3-plan.ts`)
- Wall and column **poche**, broken at the door / window openings the cut plane passes through.
- **Room tags** (number, name, area) from the rooms of that level (matched by elevation), optional boundaries.
- **Stair symbols**: flight outlines, riser lines (solid up to the cut plane, dashed above), up arrow and "UP", landings, L / U shapes.
- Railings as double lines, ramps with arrow and slope (1:12.0), curtain walls with mullion ticks and a glazing line, roofs as
  dashed outline with ridge and slope, ceilings on reflected ceiling plans only.
- **Door and window marks** (D1, W1 ... in element order, or `properties.mark`) in tag circles.
- The base generator's geometry for the six M2 types is dropped so nothing is drawn twice.

**Elevations** (`m3-project.ts`): real geometry - walls with openings, door leaves and frames, translucent glazing, sloped roofs,
stairs. Each connected piece contributes its front-facing triangles; nearer pieces hide farther ones; crease lines are drawn
where faces bend; level datums are added.

**Sections**: closed solids are cut by the plane. The cap of every cut solid is rebuilt (poche, holes punched), so oblique cuts
have the right width, a cut through a door shows only the lintel, a cut along a stair shows the stepped waist-slab profile, and
what lies beyond is drawn as projection. "Right" / "left" look directions are never mirrored. Level datums are added.

**PDF**: filled polygons are drawn with raw PDF path operators in page coordinates (`m3-pdf-fill.ts`); the old path is the fallback.
